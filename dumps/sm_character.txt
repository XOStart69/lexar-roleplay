---@param origin vector3
---@param heading number
---@param point vector3
---@return vector3
local function toLocal(origin, heading, point)
    local rad = math.rad(heading)
    local dx, dy = point.x - origin.x, point.y - origin.y

    return vec3(dx * math.cos(rad) + dy * math.sin(rad), -dx * math.sin(rad) + dy * math.cos(rad), point.z - origin.z)
end

---@param value vector3
---@return string
local function vec(value)
    return ('vec3(%.2f, %.2f, %.2f)'):format(value.x, value.y, value.z)
end

---@param view { config: table, anchor: vector3, heading: number, label: string }
---@param position vector3
---@param focus vector3
---@param fov number
local function save(view, position, focus, fov)
    local localPosition = toLocal(view.anchor, view.heading, position)
    local localFocus = toLocal(view.anchor, view.heading, focus)
    local line = ('%s = { position = %s, focus = %s, fov = %.1f },'):format(view.label, vec(localPosition), vec(localFocus), fov)

    view.config.position = localPosition
    view.config.focus = localFocus
    view.config.fov = fov

    lib.setClipboard(line)
    print(('[sm_characters] kamera: %s'):format(line))
    lib.notify({ description = 'Nilai kamera disalin ke clipboard dan F8, dipakai sampai resource restart.', type = 'success' })
end

        while freecam do
            DisableAllControlActions(0)

            yaw -= GetDisabledControlNormal(0, 1) * 6.0
            pitch = math.max(-89.0, math.min(89.0, pitch - GetDisabledControlNormal(0, 2) * 6.0))

            local rx, rz = math.rad(pitch), math.rad(yaw)
            local forward = vec3(-math.sin(rz) * math.cos(rx), math.cos(rz) * math.cos(rx), math.sin(rx))
            local right = vec3(math.cos(rz), math.sin(rz), 0.0)

            if IsDisabledControlPressed(0, 32) then position += forward * speed end
            if IsDisabledControlPressed(0, 33) then position -= forward * speed end
            if IsDisabledControlPressed(0, 35) then position += right * speed end
            if IsDisabledControlPressed(0, 34) then position -= right * speed end
            if IsDisabledControlPressed(0, 38) then position += vec3(0.0, 0.0, speed) end
            if IsDisabledControlPressed(0, 44) then position -= vec3(0.0, 0.0, speed) end
            if IsDisabledControlJustPressed(0, 241) then fov = math.max(10.0, fov - 2.0) end
            if IsDisabledControlJustPressed(0, 242) then fov = math.min(90.0, fov + 2.0) end

            SetCamCoord(cam, position.x, position.y, position.z)
            SetCamRot(cam, pitch, 0.0, yaw, 2)
            SetCamFov(cam, fov)

            if IsDisabledControlJustPressed(0, 191) then
                save(view, position, position + forward * 2.0, fov)
            end

            if IsDisabledControlJustPressed(0, 194) then
                freecam = false
            end

            Wait(0)
        end

        SetCamActive(cam, false)
        DestroyCam(cam, false)

        if DoesCamExist(base) then
            SetCamActive(base, true)
        end

        RenderScriptCams(true, false, 0, true, false)

        if select.isActive() then
            nui.focus(true)
        end
    end)

local function prefetch()
    ---@type { done: boolean, payload: table|false }
    local job = { done = false, payload = false }

    background = job

    CreateThread(function()
        local ok, payload = pcall(select.load)

        if not ok then
            lib.print.error(('gagal mengambil data karakter: %s'):format(payload))
        end

        job.payload = ok and payload or false
        job.done = true
    end)
end

---@return table|false
local function fetch()
    local job = background

    background = nil

    if job then
        pcall(lib.waitFor, function()
            return job.done or nil
        end, nil, false)

        if job.payload then return job.payload end
    end

    return select.load()
end

---@return boolean
local function open()
    local began = GetGameTimer()
    local payload = fetch()
    if not payload then return false end

    local view = select.prepare(payload)

    intro.stop()
    nui.send('loading', { visible = false })
    select.show(view)
    lib.print.info(('memuat karakter: total %d ms'):format(GetGameTimer() - began))

    return true
end

---@return boolean
local function load()
    if loading then return false end

    loading = true
    nui.send('loading', { visible = true })
    utils.fadeOut(400)
    intro.hold()

    local ok, result = pcall(open)

    loading = false

    if ok and result then return true end

    if not ok then
        lib.print.error(('gagal memuat pemilihan karakter: %s'):format(result))
    end

    if select.isActive() then
        select.close()
    end

    nui.send('loading', { visible = false, error = true })
    intro.resume()

    return false
end

local config = require 'config.client'.intro
local world = require 'modules.client.world'
local nui = require 'modules.client.nui'

local FADE_WAIT <const> = 3000

local intro = {}
local running = false
local flying = false
local alive = false
local focused = false
local session = 0
local generation = 0

local function fly()
    generation += 1
    flying = true
    alive = true

    local mine = generation

    local function live()
        return flying and mine == generation
    end

    CreateThread(function()
        local index = 1
        local shots = config.shots

        while live() do
            local shot = shots[index]
            local cam = CreateCamWithParams('DEFAULT_SCRIPTED_CAMERA', shot.from.x, shot.from.y, shot.from.z, 0.0, 0.0, 0.0, shot.fov, true, 2)

            PointCamAtCoord(cam, shot.target.x, shot.target.y, shot.target.z)
            RenderScriptCams(true, false, 0, true, false)

            if focused then
                SetFocusPosAndVel(shot.from.x, shot.from.y, shot.from.z, 0.0, 0.0, 0.0)
            end

            Wait(500)

            if live() then
                DoScreenFadeIn(1500)
            end

            local began = GetGameTimer()
            local nextFocus = 0
            local fading = false

            while live() do
                local now = GetGameTimer()
                local elapsed = now - began

                if elapsed >= shot.duration then break end

                local position = shot.from + (shot.to - shot.from) * (elapsed / shot.duration)

                SetCamCoord(cam, position.x, position.y, position.z)
                PointCamAtCoord(cam, shot.target.x, shot.target.y, shot.target.z)

                if focused and now >= nextFocus then
                    SetFocusPosAndVel(position.x, position.y, position.z, 0.0, 0.0, 0.0)
                    nextFocus = now + 250
                end

                if not fading and #shots > 1 and elapsed >= shot.duration - 1500 then
                    DoScreenFadeOut(1500)
                    fading = true
                end

                Wait(0)
            end

            if fading and live() then
                local deadline = GetGameTimer() + FADE_WAIT

                while live() and not IsScreenFadedOut() and GetGameTimer() < deadline do
                    Wait(0)
                end
            end

            DestroyCam(cam, false)
            index = index % #shots + 1
        end

        if mine ~= generation then return end

        if focused then
            ClearFocus()
        end

        alive = false
    end)
end

function intro.start()
    if running then return end

    running = true
    focused = true
    session += 1
    world.lock(config.hour, config.minute, config.weather)

    local current = session

    SetTimeout(config.uiDelay, function()
        if not running or session ~= current then return end

        nui.send('intro', { visible = true, store = config.store, credits = config.credits })
        nui.focus(true)
    end)

    fly()
end

function intro.releaseFocus()
    if not focused then return end

    focused = false
    ClearFocus()
end

function intro.hold()
    if not flying then return end

    flying = false
    intro.releaseFocus()

    pcall(lib.waitFor, function()
        return not alive or nil
    end, nil, FADE_WAIT)
end

function intro.resume()
    if not running then return intro.start() end
    if flying then return end

    focused = true
    world.lock(config.hour, config.minute, config.weather)
    fly()
end

function intro.stop()
    running = false
    nui.send('intro', { visible = false })
    intro.hold()
end

return intro

---@param ax number
---@param ay number
---@param bx number

local function cross2(ax, ay, bx, by)

end

---@param a vector3
---@param b vector3
---@return vector3
local function cross3(a, b)
    return vec3(a.y * b.z - a.z * b.y, a.z * b.x - a.x * b.z, a.x * b.y - a.y * b.x)
end

---@param raw vector3[]
---@param uv ScreenUv
---@return vector3[]
local function applyUv(raw, uv)
    local function at(u, v)
        local top = raw[1] + (raw[2] - raw[1]) * u
        local bottom = raw[4] + (raw[3] - raw[4]) * u

        return top + (bottom - top) * v
    end

    return { at(uv.left, uv.top), at(uv.right, uv.top), at(uv.right, uv.bottom), at(uv.left, uv.bottom) }
end

---@param prop number
---@param ped number
---@return PhoneScreen
local function measureScreen(prop, ped)
    local min, max = GetModelDimensions(config.model)
    local size = { max.x - min.x, max.y - min.y, max.z - min.z }
    local mid = (min + max) / 2
    local order = { 1, 2, 3 }

    table.sort(order, function(a, b) return size[a] < size[b] end)

    local thin, short, long = order[1], order[2], order[3]
    local head = GetPedBoneCoords(ped, HEAD_BONE, 0.0, 0.0, 0.0)
    local normal = unit(thin, 1.0)

    if #(toWorld(prop, mid + normal * size[thin]) - head) > #(toWorld(prop, mid - normal * size[thin]) - head) then
        normal = unit(thin, -1.0)
    end

    local up = unit(long, 1.0)

    if toWorld(prop, mid + up).z < toWorld(prop, mid - up).z then
        up = unit(long, -1.0)
    end

    local right = cross3(up, normal)
    local inset = config.screen.inset
    local center = mid + normal * (size[thin] / 2 + config.screen.offset)
    local top = size[long] / 2 - size[long] * inset.top
    local bottom = -size[long] / 2 + size[long] * inset.bottom
    local side = size[short] / 2 - size[short] * inset.x

    local raw = {
        center + up * top - right * side,
        center + up * top + right * side,
        center + up * bottom + right * side,
        center + up * bottom - right * side,
    }

    return {
        center = center,
        normal = normal,
        up = up,
        raw = raw,
        corners = config.screen.uv and applyUv(raw, config.screen.uv) or raw,
        power = applyUv(raw, config.powerButton),
    }
end

    local prop = state.prop

    if prop and DoesEntityExist(prop) then
        return state.screen and toWorld(prop, state.screen.center) or GetEntityCoords(prop)
    end

    return GetPedBoneCoords(state.ped, config.bone, 0.0, 0.0, 0.0)
end

---@param state PhoneState
---@return vector3 position
---@return vector3 target
local function framing(state)
    local view = state.view
    local target = screenTarget(state)
    local direction = GetPedBoneCoords(state.ped, HEAD_BONE, 0.0, 0.0, 0.0) - target
    local length = #direction

    if length < 0.001 then
        return target + vec3(0.0, 0.0, view.distance), target
    end

    local distance = math.min(view.distance, length * 0.8)

    return target + (direction / length) * distance + vec3(0.0, 0.0, view.lift), target
end

---@param state PhoneState
---@return number? x
---@return number? y
local function powerAnchor(state)
    if not hasScreen(state) then return end

    local area = state.screen.power
    local point = toWorld(state.prop, (area[2] + area[3]) / 2)
    local visible, x, y = GetScreenCoordFromWorldCoord(point.x, point.y, point.z)

    if visible then return x, y end
end

---@param state PhoneState
---@param durationMs integer
---@param arc boolean
local function focusScreen(state, durationMs, arc)
    local position, target = framing(state)
    local from = arc and camera.view()
    local head = GetPedBoneCoords(state.ped, HEAD_BONE, 0.0, 0.0, 0.0)
    local ahead = target - head
    local reach = math.sqrt(ahead.x * ahead.x + ahead.y * ahead.y)
    local via

    if from and reach > 0.001 then
        local forward = vec3(ahead.x / reach, ahead.y / reach, 0.0)
        local right = vec3(forward.y, -forward.x, 0.0)
        local offset = from.position - head
        local side = (offset.x * right.x + offset.y * right.y >= 0.0 and 1.0 or -1.0) * ARC_SIDE
        local beside = position + right * side + forward * ARC_LEAD + vec3(0.0, 0.0, ARC_LIFT)

        via = { (from.position + beside) / 2, position + (beside - position) * 0.8 }
    end

    camera.glide({ position = position, target = target, fov = state.view.fov, dof = { near = 0.05, far = 0.6 } }, durationMs, via)
end

---@param state PhoneState
local function run(state)
    CreateThread(function()
        local sentX, sentY = -2.0, -2.0

        while active == state do
            local x, y = powerAnchor(state)

            if math.abs((x or -1.0) - sentX) + math.abs((y or -1.0) - sentY) > 0.0008 then
                sentX, sentY = x or -1.0, y or -1.0
                nui.send('phoneAnchor', { x = x, y = y })
            end

            local poly = config.render == 'poly' and state.powered and state.dui?.txn

            if (poly or debugMode) and hasScreen(state) then
                local points = worldCorners(state)

                if poly then
                    drawScreen(points, RUNTIME_TXD, poly)
                end

                if debugMode then
                    drawOutline(points)
                    drawEdges(worldCorners(state, state.screen.power), 250, 200, 60)
                end
            end

            Wait(0)
        end
    end)
end

---@param state PhoneState
local function measure(state)
    if state.screen or not state.prop or not DoesEntityExist(state.prop) then return end

    state.screen = measureScreen(state.prop, state.ped)

    if debugMode then
        lib.print.info(('layar HP: tengah %s, normal %s, atas %s'):format(state.screen.center, state.screen.normal, state.screen.up))
    end
end

---@param state PhoneState
---@param prop number
local function attachTo(state, prop)
    local ped = state.ped

    AttachEntityToEntity(prop, ped, GetPedBoneIndex(ped, config.bone), config.offset.x, config.offset.y, config.offset.z, config.rotation.x, config.rotation.y, config.rotation.z, false, false, false, false, 2, true)
end

---@param state PhoneState
---@return boolean created
local function attachProp(state)
    if state.attaching or state.prop and DoesEntityExist(state.prop) then return false end

    state.attaching = true

    local prop = utils.createObject(config.model, GetEntityCoords(state.ped), PROP_TIMEOUT)

    state.attaching = false

    if not prop then return false end

    if state.closed then
        DeleteEntity(prop)
        return false
    end

    attachTo(state, prop)
    state.prop = prop
    state.screen = nil

    return true
end

---@param state PhoneState
---@return boolean raised
local function holdPose(state)
    if not state.animated or IsEntityPlayingAnim(state.ped, config.dict, config.enter, 3) then return false end

    TaskPlayAnim(state.ped, config.dict, config.enter, 8.0, -8.0, -1, OPEN_FLAG, 0.0, false, false, false)
    Wait(math.floor(GetAnimDuration(config.dict, config.enter) * 1000))

    return true
end

---@param state PhoneState
local function stand(state)
    local stances = config.stance
    if not stances then return end

    local stance = IsPedMale(state.ped) and stances.male or stances.female

    if not utils.loadAnimDict(stance.dict) then return end

    TaskPlayAnim(state.ped, stance.dict, stance.clip, 8.0, -8.0, -1, 1, 0.0, false, false, false)

    pcall(lib.waitFor, function()
        return IsEntityPlayingAnim(state.ped, stance.dict, stance.clip, 3) or nil
    end, nil, 500)

    SetEntityAnimSpeed(state.ped, stance.dict, stance.clip, 0.0)
    state.stance = stance
end

---@param state PhoneState
local function steady(state)
    if state.steady and state.steady() then return end

    stand(state)
end

---@param state PhoneState
local function raise(state)
    local ped = state.ped

    steady(state)
    state.animated = utils.loadAnimDict(config.dict)

    if state.animated then
        TaskPlayAnim(ped, config.dict, config.enter, 8.0, -8.0, -1, OPEN_FLAG, 0.0, false, false, false)
        Wait(300)
    end

    attachProp(state)

    if state.animated then
        Wait(math.max(GetAnimDuration(config.dict, config.enter) * 1000 - 300, 0))
    end

    measure(state)
end

---@param state PhoneState
local function createDui(state)
    duis += 1

    local url = ('https://cfx-nui-%s/web/build/phone.html?dui=%d'):format(cache.resource, duis)
    local dui = { id = duis, object = CreateDui(url, config.resolution.x, config.resolution.y), created = GetGameTimer() }

    state.pageReady = false
    state.dui = dui
    lib.print.info(('dui %d: dibuat'):format(dui.id))

    CreateThread(function()
        while state.dui == dui and not IsDuiAvailable(dui.object) do
            Wait(100)
        end

        if state.dui == dui then
            lib.print.info(('dui %d: browser tersedia (%d ms)'):format(dui.id, GetGameTimer() - dui.created))
        end
    end)
end

---@return integer
local function phoneTxd()
    runtimeTxd = runtimeTxd or CreateRuntimeTxd(RUNTIME_TXD)

    return runtimeTxd
end

---@param state PhoneState
---@param dui PhoneDui
---@return string? txn
---@return integer attempts
---@return integer elapsed
local function bindDui(state, dui)
    if dui.txn then return dui.txn, 0, 0 end

    local began = GetGameTimer()
    local attempts = 0

    repeat
        attempts += 1
        textures += 1

        local txn = ('screen_%d'):format(textures)
        local texture = CreateRuntimeTextureFromDuiHandle(phoneTxd(), txn, GetDuiHandle(dui.object))

        if texture and texture ~= 0 and GetRuntimeTextureWidth(texture) > 0 then
            dui.txn = txn

            return txn, attempts, GetGameTimer() - began
        end

        Wait(BIND_INTERVAL)
    until active ~= state or GetGameTimer() - began >= BIND_TIMEOUT

    return nil, attempts, GetGameTimer() - began
end

---@param dui PhoneDui
---@param message table
local function postDui(dui, message)
    SendDuiMessage(dui.object, json.encode(message))
end

---@param state PhoneState
local function screenOff(state)
    textures += 1

    local txn = ('off_%d'):format(textures)
    local texture = CreateRuntimeTexture(phoneTxd(), txn, OFF_SIZE, OFF_SIZE)

    for x = 0, OFF_SIZE - 1 do
        for y = 0, OFF_SIZE - 1 do
            SetRuntimeTexturePixel(texture, x, y, OFF_COLOR[1], OFF_COLOR[2], OFF_COLOR[3], 255)
        end
    end

    CommitRuntimeTexture(texture)
    RemoveReplaceTexture(config.txd, config.txn)
    AddReplaceTexture(config.txd, config.txn, RUNTIME_TXD, txn)
    state.replaced = true
end

---@param state PhoneState
local function restoreScreen(state)
    if not state.replaced then return end

    RemoveReplaceTexture(config.txd, config.txn)
    state.replaced = false
end

---@param state PhoneState
local function destroyDui(state)
    local dui = state.dui
    if not dui then return end

    state.dui = nil
    SetDuiUrl(dui.object, 'about:blank')
    DestroyDui(dui.object)
    lib.print.info(('dui %d: dihapus'):format(dui.id))
end

---@param state PhoneState
local function sendLocations(state)
    if not state.dui then return end

    postDui(state.dui, { action = 'locations', data = { locations = state.list, character = state.info } })
end

---@param state PhoneState
---@param cooldown? integer
local function sendStatus(state, cooldown)
    nui.send('phoneStatus', { powered = state.powered, cooldown = cooldown })
end

---@param state PhoneState
---@param check fun(): boolean
---@param timeout integer
---@return boolean
local function waitUntil(state, check, timeout)
    pcall(lib.waitFor, function()
        if active ~= state or check() then return true end
    end, nil, timeout)

    return active == state and check()
end

---@param state PhoneState
local function showProp(state)
    if not state.powered then
        screenOff(state)
    elseif state.dui?.txn and config.render == 'texture' then
        RemoveReplaceTexture(config.txd, config.txn)
        AddReplaceTexture(config.txd, config.txn, RUNTIME_TXD, state.dui.txn)
        state.replaced = true
    end

    Wait(100)
    measure(state)

    if active ~= state then return end

    focusScreen(state, REFOCUS_MS, false)
    nui.send('phone', { visible = true, fallback = not hasScreen(state) })
end

---@param state PhoneState
---@param attempts integer
---@return integer attempts
local function checkProp(state, attempts)
    if active ~= state or state.attaching then return attempts end

    local prop = state.prop

    if prop and DoesEntityExist(prop) then
        if not IsEntityAttachedToEntity(prop, state.ped) then
            lib.print.warn('prop hp: lepas dari tangan, dipasang ulang')
            attachTo(state, prop)
        end

        return 0
    end

    if prop then
        state.prop = nil
        lib.print.warn('prop hp: hilang, dibuat ulang')
    end

    attempts += 1

    if not attachProp(state) then
        if active == state then
            lib.print.warn(('prop hp: gagal dibuat (percobaan %d)'):format(attempts))
        end

        return attempts
    end

    lib.print.info(('prop hp: dibuat (percobaan %d)'):format(attempts))
    showProp(state)

    return 0
end

---@param state PhoneState
local function keepProp(state)
    CreateThread(function()
        local attempts = 0

        while active == state do
            Wait(PROP_RETRY_MS)
            attempts = checkProp(state, attempts)
        end
    end)
end

---@param state PhoneState
local function render(state)
    state.renders += 1

    local renders = state.renders

    if not state.animated then
        state.animated = utils.loadAnimDict(config.dict)
    end

    local moved = holdPose(state)

    if attachProp(state) then
        moved = true
        screenOff(state)
        Wait(100)
    end

    measure(state)

    if moved and active == state then
        focusScreen(state, REFOCUS_MS, false)
    end

    if not state.dui then
        createDui(state)
    end

    local dui = state.dui --[[@as PhoneDui]]

    if not waitUntil(state, function() return IsDuiAvailable(dui.object) end, DUI_TIMEOUT) then
        if active ~= state then return end

        lib.print.error(('dui %d: browser tidak tersedia (%d ms), dibatalkan'):format(dui.id, GetGameTimer() - dui.created))
        destroyDui(state)
        state.powered = false
        return
    end

    if not waitUntil(state, function() return state.pageReady end, PAGE_TIMEOUT) then
        if active ~= state then return end

        lib.print.warn(('dui %d: halaman belum siap (%d ms), texture tetap diikat'):format(dui.id, GetGameTimer() - dui.created))
    end

    Wait(FIRST_RENDER_MS)

    if active ~= state then return end

    local txn, attempts, elapsed = bindDui(state, dui)

    if active ~= state then return end

    if not txn then
        lib.print.error(('dui %d: texture tidak terikat (%d percobaan, %d ms), dibatalkan'):format(dui.id, attempts, elapsed))
        destroyDui(state)
        state.powered = false
        return
    end

    if attempts > 0 then
        local log = attempts > 1 and lib.print.warn or lib.print.info

        log(('dui %d: texture terikat (percobaan %d, %d ms)'):format(dui.id, attempts, elapsed))
    end

    if config.render == 'texture' then
        RemoveReplaceTexture(config.txd, config.txn)
        AddReplaceTexture(config.txd, config.txn, RUNTIME_TXD, txn)
        state.replaced = true
    end

    lib.print.info(('dui %d: layar on'):format(dui.id))
    state.powered = true
    postDui(dui, { action = 'power' })
    sendLocations(state)

    for i = 1, #RESEND_DELAYS do
        SetTimeout(RESEND_DELAYS[i], function()
            if active == state and state.renders == renders then
                sendLocations(state)
            end
        end)
    end

    nui.send('phone', { visible = true, fallback = not hasScreen(state) })
end

---@param state PhoneState
---@return boolean
local function power(state)
    if state.rendering then return false end

    state.rendering = true

    local ok, result = pcall(render, state)

    state.rendering = false

    if not ok then
        lib.print.error(('gagal render handphone: %s'):format(result))
    end

    if active == state then
        sendStatus(state)
    end

    return ok
end

---@param state PhoneState
local function powerOff(state)
    state.renders += 1
    state.powered = false
    calibration = nil
    screenOff(state)
    lib.print.info(('dui %d: layar off'):format(state.dui and state.dui.id or 0))
    destroyDui(state)
    createDui(state)
    sendStatus(state)
end

---@param corners vector3[]
---@param x number
---@param y number
---@return number? u
---@return number? v
local function cursorUv(corners, x, y)
    local points = {}

    for i = 1, 4 do
        local visible, sx, sy = GetScreenCoordFromWorldCoord(corners[i].x, corners[i].y, corners[i].z)

        if not visible then return end

        points[i] = { x = sx, y = sy }
    end

    local p0, p1, p2, p3 = points[1], points[2], points[3], points[4]
    local u, v = 0.5, 0.5

    for _ = 1, 10 do
        local iu, iv = 1 - u, 1 - v
        local qx = iu * iv * p0.x + u * iv * p1.x + u * v * p2.x + iu * v * p3.x
        local qy = iu * iv * p0.y + u * iv * p1.y + u * v * p2.y + iu * v * p3.y
        local ex, ey = x - qx, y - qy

        if math.abs(ex) < 0.00001 and math.abs(ey) < 0.00001 then break end

        local dux = -iv * p0.x + iv * p1.x + v * p2.x - v * p3.x
        local duy = -iv * p0.y + iv * p1.y + v * p2.y - v * p3.y
        local dvx = -iu * p0.x - u * p1.x + u * p2.x + iu * p3.x
        local dvy = -iu * p0.y - u * p1.y + u * p2.y + iu * p3.y
        local det = cross2(dux, duy, dvx, dvy)

        if math.abs(det) < 0.0000001 then break end

        u = u + cross2(ex, ey, dvx, dvy) / det
        v = v + cross2(dux, duy, ex, ey) / det
    end

    return u, v
end

---@param x number
---@param y number
---@return number? u
---@return number? v
local function screenToUv(x, y)
    local u, v = cursorUv(worldCorners(active --[[@as PhoneState]]), x, y)

    if not u or u < 0 or u > 1 or v < 0 or v > 1 then return end

    if config.render == 'texture' then
        if config.screen.flipX then u = 1 - u end
        if config.screen.flipY then v = 1 - v end
    end

    return u, v
end

---@param state PhoneState
---@param x number
---@param y number
---@return boolean
local function onPowerButton(state, x, y)
    local corners = worldCorners(state, state.screen.power)
    local points = {}

    for i = 1, 4 do
        local visible, sx, sy = GetScreenCoordFromWorldCoord(corners[i].x, corners[i].y, corners[i].z)

        if not visible then return false end

        points[i] = { x = sx, y = sy }
    end

    local sign = 0

    for i = 1, 4 do
        local a, b = points[i], points[i % 4 + 1]
        local side = cross2(b.x - a.x, b.y - a.y, x - a.x, y - a.y)
        local current = side > 0 and 1 or side < 0 and -1 or 0

        if current ~= 0 then
            if sign ~= 0 and current ~= sign then return false end

            sign = current
        end
    end

    return true
end

---@param state PhoneState
---@return boolean ok
---@return integer wait
local function tryPower(state)
    local now = GetGameTimer()
    local wait = state.poweredAt and state.poweredAt + config.powerCooldown - now or 0

    if wait > 0 or state.rendering then return false, math.max(wait, 0) end

    state.poweredAt = now
    sendStatus(state, config.powerCooldown)

    CreateThread(function()
        if state.powered then
            powerOff(state)
        else
            power(state)
        end
    end)

    return true, config.powerCooldown
end

---@param state PhoneState
---@param step integer
local function showCalibration(state, step)
    postDui(state.dui --[[@as PhoneDui]], { action = 'calibrate', data = { step = step, targets = CALIBRATION } })
end

---@param x number
---@param y number
local function calibrateClick(x, y)
    local state = active --[[@as PhoneState]]
    local u, v = cursorUv(worldCorners(state, state.screen.raw), x, y)
    if not u or not calibration then return end

    calibration[#calibration + 1] = { u = u, v = v }

    if #calibration < #CALIBRATION then
        return showCalibration(state, #calibration + 1)
    end

    local first, second = calibration[1], calibration[2]
    local a, b = CALIBRATION[1], CALIBRATION[2]
    local width = (second.u - first.u) / (b[1] - a[1])
    local height = (second.v - first.v) / (b[2] - a[2])

    calibration = nil
    showCalibration(state, 0)

    if math.abs(width) < 0.05 or math.abs(height) < 0.05 then
        return lib.print.warn('kalibrasi layar HP gagal: dua klik terlalu berdekatan, ulangi /smchar_phonecal')
    end

    ---@type ScreenUv
    local uv = { left = first.u - a[1] * width, top = first.v - a[2] * height }

    uv.right = uv.left + width
    uv.bottom = uv.top + height
    state.screen.corners = applyUv(state.screen.raw, uv)

    lib.print.info(('kalibrasi layar HP: uv = { left = %.4f, top = %.4f, right = %.4f, bottom = %.4f }'):format(uv.left, uv.top, uv.right, uv.bottom))
end

---@param list table[]
---@param info table
---@param onPick fun(id: string)
---@param options? PhoneOptions
function phone.open(list, info, onPick, options)
    options = options or {}

    ---@type PhoneState
    local state = {
        ped = PlayerPedId(),
        steady = options.steady,
        view = options.child and config.childCamera or config.camera,
        list = list,
        info = info,
        onPick = onPick,
        animated = false,
        pageReady = false,
        replaced = false,
        powered = false,
        rendering = false,
        renders = 0,
        attaching = false,
        closed = false,
    }

    screenOff(state)
    raise(state)
    screenOff(state)

    active = state
    createDui(state)
    focusScreen(state, GLIDE_MS, true)
    run(state)
    keepProp(state)

    nui.send('phone', { visible = true, fallback = not hasScreen(state) })
    sendStatus(state)
    nui.focus(true)
end

---@param list table[]
---@param info table
function phone.rerender(list, info)
    local state = active
    if not state then return end

    state.list = list
    state.info = info

    if state.powered then
        sendLocations(state)
    end
end

---@param list table[]
---@param info table
---@return boolean
function phone.render(list, info)
    local state = active
    if not state or state.rendering then return false end

    state.list = list
    state.info = info

    if state.powered then
        powerOff(state)
    end

    return power(state)
end

---@return boolean
function phone.isOpen()
    return active ~= nil
end

function phone.preload()
    CreateThread(function()
        utils.loadModel(config.model, PRELOAD_TIMEOUT)
        utils.loadAnimDict(config.dict, PRELOAD_TIMEOUT)

        if config.stance then
            utils.loadAnimDict(config.stance.male.dict, PRELOAD_TIMEOUT)
            utils.loadAnimDict(config.stance.female.dict, PRELOAD_TIMEOUT)
        end
    end)
end

---@return boolean
function phone.calibrate()
    if not active or not active.powered or not hasScreen(active) then return false end

    calibration = {}
    showCalibration(active, 1)

    return true
end

---@param instant? boolean
function phone.close(instant)
    local state = active
    if not state then return end

    active = nil
    calibration = nil
    state.closed = true

    nui.send('phone', { visible = false })
    nui.focus(false)
    restoreScreen(state)
    destroyDui(state)

    if not instant and state.animated then
        TaskPlayAnim(state.ped, config.dict, config.exit, 8.0, -8.0, 950, CLOSE_FLAG, 0.0, false, false, false)
        Wait(400)
    end

    if state.prop and DoesEntityExist(state.prop) then
        DeleteEntity(state.prop)
    end

    if state.stance then
        StopAnimTask(state.ped, state.stance.dict, state.stance.clip, 2.0)
        RemoveAnimDict(state.stance.dict)
    end

    RemoveAnimDict(config.dict)
end

RegisterNUICallback('phonePower', function(_, cb)
    if not active then return cb({ ok = false }) end

    local ok, wait = tryPower(active)

    cb({ ok = ok, wait = wait })
end)

RegisterNUICallback('phoneReady', function(data, cb)
    cb(1)

    local dui = active and active.dui
    local id = type(data) == 'table' and math.tointeger(data.dui)

    if not dui or id ~= dui.id then return end

    active.pageReady = true
    lib.print.info(('dui %d: halaman siap (%d ms)'):format(dui.id, GetGameTimer() - dui.created))

    if active.powered then
        postDui(dui, { action = 'power' })
    end

    sendLocations(active)
end)

RegisterNUICallback('phoneMouse', function(data, cb)
    cb(1)

    if not active or not hasScreen(active) or type(data) ~= 'table' then return end

    if data.type == 'down' and data.button ~= 2 then
        local x, y = tonumber(data.x) or -1, tonumber(data.y) or -1
        local hit = onPowerButton(active, x, y)

        if debugMode then
            lib.print.info(('klik %.3f, %.3f: tombol power %s'):format(x, y, hit and 'kena' or 'tidak kena'))
        end

        if hit then
            tryPower(active)
            return
        end
    end

    if not active.powered then return end

    if calibration and data.type ~= 'move' then
        if data.type == 'down' and data.button ~= 2 then
            calibrateClick(tonumber(data.x) or -1, tonumber(data.y) or -1)
        end

        return
    end

    local u, v = screenToUv(tonumber(data.x) or -1, tonumber(data.y) or -1)

    if debugMode and data.type == 'down' then

local function walkAway(ped, partner)
    local leave = selection.partnerLeave
    local origin = GetEntityCoords(partner)
    local side = GetOffsetFromEntityGivenWorldCoords(ped, origin.x, origin.y, origin.z).x < 0.0 and -1.0 or 1.0
    local away = GetOffsetFromEntityInWorldCoords(ped, side * leave.side, -leave.back, 0.0)
    local heading = GetHeadingFromVector_2d(away.x - origin.x, away.y - origin.y)
    local deadline = GetGameTimer() + leave.waitMs

    ClearPedTasks(partner)
    TaskGoStraightToCoord(partner, away.x, away.y, away.z, 1.0, leave.waitMs * 2, heading, 0.5)

    CreateThread(function()
        while DoesEntityExist(partner) do
            local far = #(GetEntityCoords(partner) - GetEntityCoords(ped)) >= leave.removeDistance

            if far or IsPedFalling(partner) or GetGameTimer() >= deadline then
                FreezeEntityPosition(partner, true)
                DeleteEntity(partner)
                break
            end

            Wait(0)
        end
    end)
end

---@param character table
---@param child table?
---@param partner table?
local function prefetch(character, child, partner)
    requestModel(skinFor(character.citizenid, character.gender).model)

    if child then
        RequestAnimDict(child.dict)
        return
    end

    if not partner then
        local drinking = selection.drinking

        RequestAnimDict(drinking.dict)

        for i = 1, #drinking.props do
            requestModel(drinking.props[i].model)
        end

        return
    end

    if waitSkin(partner.citizenid, SKIN_TIMEOUT) then
        requestModel(skinFor(partner.citizenid, partner.gender).model)
    end

    local scenes = selection.duoScene[partner.type] == 'bedroom' and selection.bedroom or selection.couple[partner.type]

    for i = 1, #scenes do
        RequestAnimDict(scenes[i].dict)
    end
end

---@param character table
---@return CameraFocus
local function buildPreview(character)
    clearStage()

    if not waitSkin(character.citizenid, SKIN_TIMEOUT) then
        lib.print.warn(('memuat karakter: skin %s belum diterima, pakai default'):format(character.citizenid))
    end

    local child = character.child and selection.child or nil
    local partner = not child and character.partner or nil

    if partner?.child then
        partner = nil
    end

    local began = GetGameTimer()

    prefetch(character, child, partner)

    local ped = utils.applyPlayer(skinFor(character.citizenid, character.gender), character.gender)
    local modelAt = GetGameTimer()
    local scene = partner and coupleScene(partner)
    local partnerAt = GetGameTimer()
    local bed = scene and scene.bed
    local spot = child and child.coords or bed and bed.coords or spotFor(character, scene ~= nil)

    utils.teleport(ped, spot, true)
    ResetEntityAlpha(ped)
    SetEntityVisible(ped, true, false)

    local placedAt = GetGameTimer()

    if child then
        stageChild(ped, child)
    elseif bed then
        stageBed(ped, bed, partner)
    elseif scene then
        stageCouple(ped, spot, partner, scene.anim)
    else
        stageDrinking(ped, spot)
    end

    lib.print.info(('memuat karakter: model %d ms (%s), pasangan %d ms, lokasi %d ms, pose %d ms'):format(modelAt - began, tostring(skinFor(character.citizenid, character.gender).model), partnerAt - modelAt, placedAt - partnerAt, GetGameTimer() - placedAt))
    Wait(250)

    if child then
        return sceneView(ped, GetEntityCoords(ped), child.coords.w, child.camera)
    elseif bed then
        return sceneView(ped, bed.coords.xyz, bed.coords.w, bed.camera)
    end

    return viewFor(spot, anchorOf(ped), scene ~= nil)
end

---@return table|false
function select.load()
    received = nil
    table.wipe(skins)
    table.wipe(decoded)

    local began = GetGameTimer()

    TriggerServerEvent('sm_characters:server:fetch')

    local payload = net.awaitPush(function() return received end, FETCH_TIMEOUT)

    if not payload then
        lib.print.warn(('memuat karakter: data %s (%d ms)'):format(payload == false and 'ditolak server' or 'tidak diterima', GetGameTimer() - began))
        return false
    end

    lib.print.info(('memuat karakter: data diterima (%d ms)'):format(GetGameTimer() - began))

    return payload
end

---@param kind string
---@param payload table
function select.sync(kind, payload)
    if not state.active then return end

    if kind == 'incoming' then
        state.incoming[#state.incoming + 1] = payload
    elseif kind == 'incomingRemoved' then
        removeWhere(state.incoming, function(item) return item.id == payload.id end)
    elseif kind == 'outgoingRemoved' then
        removeWhere(state.outgoing, function(item) return item.id == payload.id end)
    elseif kind == 'linked' then
        local character = findCharacter(payload.citizenid)

        if character then
            character.partner = payload.partner
            removeWhere(state.outgoing, function(item) return item.sender == payload.citizenid end)
        end
    elseif kind == 'unlinked' then
        local character = findCharacter(payload.citizenid)

        if character then
            character.partner = nil
        end
    end

    pushUpdate()

    if kind == 'linked' or kind == 'unlinked' then
        refreshIfCurrent(payload.citizenid)
    end
end

    state.busy = true

        birthdate = data.birthdate,

        nationality = data.nationality,

    local character = state.active and not state.busy and findCharacter(data?.citizenid)

RegisterNUICallback('inviteRespond', function(data, cb)
    local id = math.tointeger(data?.id)
    if not id then return cb({ error = 'invalid' }) end

    local result = lib.callback.await('sm_characters:partner:respond', false, id, data.accept == true)

    if result?.ok and state.active then
        removeWhere(state.incoming, function(item) return item.id == id end)

        local character = result.partner and findCharacter(result.citizenid)

        if character then
            character.partner = result.partner

            removeWhere(state.incoming, function(item)
                return item.target == result.citizenid or item.sender == result.partner.citizenid
            end)

            removeWhere(state.outgoing, function(item) return item.sender == result.citizenid end)
        end

        pushUpdate()

        if character then
            refreshIfCurrent(result.citizenid)
        end
    end

    cb(result or { error = 'failed' })
end)

---@param event string
---@param player number
---@param ... any
function Net.emitLatentClient(event, player, ...)
    local payload = msgpack.pack_args(...)
    local payloadLen = #payload
    local bps = payloadLen + 1000
    TriggerLatentClientEventInternal(event, player --[[@as string]], payload, payloadLen, bps)
end

---@generic T
---@param get fun(): T?
---@param timeout number?
---@return T?
function Net.awaitPush(get, timeout)
    local value = get()

    if value ~= nil then return value end

    local deadline = GetGameTimer() + (timeout or 5000)

    repeat
        Wait(0)
        value = get()
    until value ~= nil or GetGameTimer() > deadline

    return value
end

---@class CameraView
---@field position vector3
---@field target vector3
---@field fov number
---@field dof { near: number, far: number }|false

---@type number?
local active
---@type CameraView?
local view
local token = 0

---@param t number
---@return number
local function smooth(t)
    if t >= 1.0 then return 1.0 end

    return t * t * (3.0 - 2.0 * t)
end

---@param from vector3
---@param first vector3
---@param second vector3
---@param to vector3
---@param t number
---@return vector3
local function bezier(from, first, second, to, t)
    local u = 1.0 - t

    return from * (u * u * u) + first * (3.0 * u * u * t) + second * (3.0 * u * t * t) + to * (t * t * t)
end

---@param position vector3
---@param fov number
---@return number
local function create(position, fov)
    return CreateCamWithParams('DEFAULT_SCRIPTED_CAMERA', position.x, position.y, position.z, 0.0, 0.0, 0.0, fov, false, 2)
end

---@param cam number
---@param easeMs number
local function activate(cam, easeMs)
    local previous = active

    active = cam
    token += 1

    if previous and DoesCamExist(previous) and easeMs > 0 then
        SetCamActiveWithInterp(cam, previous, easeMs, 1, 1)
        SetTimeout(easeMs + 100, function()
            DestroyCam(previous, false)
        end)
    else
        SetCamActive(cam, true)
        RenderScriptCams(true, false, 0, true, false)

        if previous then
            DestroyCam(previous, false)
        end
    end

    return token
end

---@class CameraFocus
---@field position vector3
---@field target vector3
---@field fov number
---@field dof? { near: number, far: number }|false
---@field easeMs? number

---@param data CameraFocus
---@return promise
function camera.focus(data)
    local done = promise.new()
    local cam = create(data.position, data.fov)
    local dof = data.dof ~= false and (data.dof or DEFAULT_DOF) or false

    PointCamAtCoord(cam, data.target.x, data.target.y, data.target.z)

    if dof then
        SetCamUseShallowDofMode(cam, true)
        SetCamNearDof(cam, dof.near)
        SetCamFarDof(cam, dof.far)
        SetCamDofStrength(cam, 1.0)
    end

    local id = activate(cam, data.easeMs or 0)

    view = { position = data.position, target = data.target, fov = data.fov, dof = dof }
    done:resolve(true)

    if dof then
        CreateThread(function()
            while id == token and DoesCamExist(cam) do
                SetUseHiDof()
                Wait(0)
            end
        end)
    end

    return done
end

---@param data CameraFocus
---@param durationMs integer
---@param via? { [1]: vector3, [2]: vector3 }
function camera.glide(data, durationMs, via)
    local cam, from = active, view

    if not cam or not DoesCamExist(cam) or not from or durationMs <= 0 then
        camera.focus(data)
        return
    end

    token += 1

    local id = token
    local dof = data.dof ~= false and (data.dof or DEFAULT_DOF) or false
    local fromDof = from.dof or dof
    local began = GetGameTimer()

    view = { position = data.position, target = data.target, fov = data.fov, dof = dof }

    if dof then
        SetCamUseShallowDofMode(cam, true)
        SetCamDofStrength(cam, 1.0)
    end

    CreateThread(function()
        while id == token and DoesCamExist(cam) do
            local t = smooth(math.min(1.0, (GetGameTimer() - began) / durationMs))
            local position = via and bezier(from.position, via[1], via[2], data.position, t) or from.position + (data.position - from.position) * t
            local target = from.target + (data.target - from.target) * t

            SetCamCoord(cam, position.x, position.y, position.z)
            PointCamAtCoord(cam, target.x, target.y, target.z)
            SetCamFov(cam, from.fov + (data.fov - from.fov) * t)

            if dof and fromDof then
                SetCamNearDof(cam, fromDof.near + (dof.near - fromDof.near) * t)
                SetCamFarDof(cam, fromDof.far + (dof.far - fromDof.far) * t)
                SetUseHiDof()
            end

            Wait(0)
        end
    end)
end

    return view
end

---@param from vector3
---@param point vector3
---@param amount number
---@return vector3
function camera.aside(from, point, amount)
    local direction = point - from
    local length = math.sqrt(direction.x * direction.x + direction.y * direction.y)

    if amount == 0.0 or length < 0.001 then return point end

    return point - vec3(direction.y / length, -direction.x / length, 0.0) * amount
end

---@param from vector3
---@param to vector3
---@param ignore number
---@return vector3
local function reachable(from, to, ignore)
    local path = to - from
    local length = #path

    if length < 0.001 then return to end

    local direction = path / length
    local reach = to + direction * FOLLOW_MARGIN
    local probe = StartExpensiveSynchronousShapeTestLosProbe(from.x, from.y, from.z, reach.x, reach.y, reach.z, FOLLOW_FLAGS, ignore, 4)
    local _, hit, coords = GetShapeTestResult(probe)

    if hit ~= true and hit ~= 1 then return to end

    local free = #(coords - from) - FOLLOW_MARGIN

    return free > 0.0 and from + direction * math.min(free, length) or from
end

---@param from vector3
---@param point vector3
---@param angle number
---@return vector3
local function turned(from, point, angle)
    local direction = point - from
    local cos, sin = math.cos(angle), math.sin(angle)

    return from + vec3(direction.x * cos - direction.y * sin, direction.x * sin + direction.y * cos, direction.z)
end

---@param ped number
---@param bone integer
---@param shift? number
function camera.follow(ped, bone, shift)
    local cam = active
    if not cam then return end

    token += 1

    local id = token

    CreateThread(function()
        local position = GetCamCoord(cam)
        local offset = position - GetEntityCoords(ped)
        local target = GetPedBoneCoords(ped, bone, 0.0, 0.0, 0.0)
        local flat = target - position
        local angle = math.atan(shift or 0.0, math.sqrt(flat.x * flat.x + flat.y * flat.y))
        local began = GetGameTimer()

        while id == token and DoesCamExist(cam) and DoesEntityExist(ped) do
            local frame = GetFrameTime()
            local desired = reachable(position, GetEntityCoords(ped) + offset, ped)
            local recenter = smooth(math.min(1.0, (GetGameTimer() - began) / FOLLOW_RECENTER_MS))

            position = position + (desired - position) * math.min(1.0, frame * 1.5)
            target = target + (GetPedBoneCoords(ped, bone, 0.0, 0.0, 0.0) - target) * math.min(1.0, frame * 3.0)

            local aim = turned(position, target, angle * (1.0 - recenter))

            if view then
                view.position = position
                view.target = aim
            end

            SetCamCoord(cam, position.x, position.y, position.z)
            PointCamAtCoord(cam, aim.x, aim.y, aim.z)
            SetUseHiDof()
            Wait(0)
        end
    end)
end

---@param easeMs? number
function camera.destroy(easeMs)
    token += 1
    view = nil

    if not active then return end

    RenderScriptCams(false, (easeMs or 0) > 0, easeMs or 0, true, false)
    DestroyCam(active, false)
    active = nil
end

---@return number?
function camera.handle()
    return active
end

return camera

        Wait(0)

    if interior == 0 then return end

    timeout = GetGameTimer() + WORLD_TIMEOUT

    while not IsInteriorReady(interior) and GetGameTimer() < timeout do
        Wait(0)
    end
end

local function buildInvites(ids, byCitizen)
    local incoming, outgoing = {}, {}

    if #ids == 0 then return incoming, outgoing end

    local placeholders = {}
    for i = 1, #ids do placeholders[i] = '?' end
    placeholders = table.concat(placeholders, ',')

    local params = {}
    for i = 1, #ids do params[#params + 1] = ids[i] end
    for i = 1, #ids do params[#params + 1] = ids[i] end
    params[#params + 1] = os.time()

    local rows = MySQL.query.await(([[
        SELECT i.citizenid, i.target, i.expires,
               p1.charinfo AS frominfo, p2.charinfo AS toinfo
        FROM character_invites i
        LEFT JOIN players p1 ON p1.citizenid = i.citizenid
        LEFT JOIN players p2 ON p2.citizenid = i.target
        WHERE (i.citizenid IN (%s) OR i.target IN (%s)) AND i.expires > ?
    ]]):format(placeholders, placeholders), params) or {}

    for _, row in ipairs(rows) do
        local mine = byCitizen[row.target]
        local theirs = byCitizen[row.citizenid]

        if mine then
            incoming[#incoming + 1] = {
                citizenid = row.citizenid,
                name = fullName(decode(row.frominfo)),
                expires = tonumber(row.expires) or 0,
            }
        end

        if theirs then
            outgoing[#outgoing + 1] = {
                citizenid = row.target,
                name = fullName(decode(row.toinfo)),
                expires = tonumber(row.expires) or 0,
            }
        end
    end

    return incoming, outgoing
end

local function sanitizeName(value)
    if type(value) ~= 'string' then return nil end
    value = value:gsub('^%s*(.-)%s*$', '%1')
    if #value < 2 or #value > 20 then return nil end
    if not value:match('^[%a][%a%s%-]*$') then return nil end
    return value
end

local MONTHS = {
    januari = 1, februari = 2, maret = 3, april = 4, mei = 5, juni = 6, juli = 7,
    agustus = 8, september = 9, oktober = 10, november = 11, desember = 12,
    jan = 1, feb = 2, mar = 3, apr = 4, jun = 6, jul = 7, agu = 8, ags = 8, sep = 9, okt = 10, nov = 11, des = 12,
    january = 1, february = 2, march = 3, may = 5, june = 6, july = 7, august = 8, october = 10, december = 12,
}

local function sanitizeBirthdate(value)
    if type(value) ~= 'string' then return nil end
    value = value:gsub('^%s*(.-)%s*$', '%1')

    local y, m, d
    local a, b, c = value:match('^(%d%d%d%d)%-(%d%d)%-(%d%d)$')

    if a then
        y, m, d = tonumber(a), tonumber(b), tonumber(c)
    else
        local day, mon, year = value:match('^(%d%d?)%s+([%a]+)%s+(%d%d%d%d)$')
        if not day then return nil end
        m = MONTHS[mon:lower()]
        if not m then return nil end
        y, d = tonumber(year), tonumber(day)
    end

    if not y or y < 1900 or y > 2015 then return nil end
    if m < 1 or m > 12 then return nil end
    if not d or d < 1 or d > 31 then return nil end

    return ('%04d-%02d-%02d'):format(y, m, d)
end

local function sanitizeGender(value)
    if value == true or value == 1 or value == 'male' or value == 'm' or value == 'M' then return 'male' end
    if value == false or value == 0 or value == 'female' or value == 'f' or value == 'F' then return 'female' end
    return nil
end

    if not lastname then return false, 'Nama belakang tidak valid (2-20 huruf)' end

    local gender = sanitizeGender(input.gender)
    if not gender then return false, 'Jenis kelamin tidak valid' end

    local birthdate = sanitizeBirthdate(input.birthdate)
    if not birthdate then return false, 'Tanggal lahir harus YYYY-MM-DD' end

    local slots = getSlots(license)
    local count = MySQL.scalar.await('SELECT COUNT(*) FROM players WHERE license = ?', { license }) or 0
    if count >= slots then return false, ('Slot karakter penuh (%d/%d)'):format(count, slots) end

    local citizenid = QBCore.Player.CreateCitizenId()
    local spawn = DEFAULT_SPAWN
    local nationality = type(input.nationality) == 'string' and input.nationality:sub(1, 32) or 'Indonesia'

    local charinfo = {
        firstname = firstname,
        lastname = lastname,
        birthdate = birthdate,
        gender = gender,
        nationality = nationality,
        cid = math.random(10000, 99999),
        phone = generatePhone(),
    }

    local money = { cash = 500, bank = 5000, crypto = 0 }
    local job = { name = 'unemployed', label = 'Pengangguran', payment = 10, onduty = true, isboss = false, grade = { name = 'Freelancer', level = 0 } }
    local gang = { name = 'none', label = 'Tidak ada', grade = { name = 'Tidak ada', level = 0 } }
    local metadata = { callsign = 'NO CALLSIGN', licences = {} }

    local nextId = (MySQL.scalar.await('SELECT COALESCE(MAX(id), 0) + 1 FROM players') or 1)

    local params = {
        nextId,
        citizenid,
        charinfo.cid,
        license,
        ('%s %s'):format(firstname, lastname),
        json.encode(money),
        json.encode(charinfo),
        json.encode(job),
        json.encode(gang),
        json.encode(spawn),
        json.encode(metadata),
    }

    local ok, err = pcall(function()
        MySQL.insert.await([[
            INSERT INTO players (id, citizenid, cid, license, name, money, charinfo, job, gang, position, metadata)
            VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)
        ]], params)
    end)

    if not ok then
        -- Skema tanpa kolom cid/name.
        ok, err = pcall(function()
            MySQL.insert.await([[
                INSERT INTO players (id, citizenid, license, money, charinfo, job, gang, position, metadata)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?)
            ]], { nextId, citizenid, license, params[6], params[7], params[8], params[9], params[10], params[11] })
        end)
    end

    if not ok then
        lib.print.error(('gagal membuat karakter: %s'):format(err))
        return false, 'Gagal menyimpan karakter'
    end

    return { coords = vec4(spawn.x, spawn.y, spawn.z, spawn.w), citizenid = citizenid }
end)

lib.callback.register('sm_characters:select', function(source, citizenid)
    local license = getLicense(source)
    if not license then return false, 'Lisensi tidak terbaca' end

    if QBCore.Functions.GetPlayer(source) then
        return false, 'Karakter sudah dimuat'
    end

    local owner = MySQL.scalar.await('SELECT license FROM players WHERE citizenid = ?', { citizenid })
    if not owner then return false, 'Karakter tidak ditemukan' end
    if owner ~= license then return false, 'Karakter bukan milik Anda' end

    local ok, result = pcall(QBCore.Player.Login, source, citizenid)
    if not ok then
        lib.print.error(('login gagal: %s'):format(result))
        return false, 'Gagal memuat karakter'
    end

    if not result then return false, 'Gagal memuat karakter' end

    return true, 'Karakter dimuat'
end)

lib.callback.register('sm_characters:delete', function(source, citizenid)
    local license = getLicense(source)
    if not license then return false, 'Lisensi tidak terbaca' end

    local owner = MySQL.scalar.await('SELECT license FROM players WHERE citizenid = ?', { citizenid })
    if not owner then return false, 'Karakter tidak ditemukan' end
    if owner ~= license then return false, 'Karakter bukan milik Anda' end

    local total = MySQL.scalar.await('SELECT COUNT(*) FROM players WHERE license = ?', { license }) or 0
    if total <= 1 then return false, 'Karakter terakhir tidak bisa dihapus' end

    local player = QBCore.Functions.GetPlayer(source)
    if player and player.PlayerData.citizenid == citizenid then
        TriggerEvent('QBCore:Server:OnPlayerUnload', source)
        QBCore.Players[source] = nil
    end

    MySQL.query.await('DELETE FROM players WHERE citizenid = ?', { citizenid })
    MySQL.query.await('DELETE FROM playerskins WHERE citizenid = ?', { citizenid })
    MySQL.query.await('DELETE FROM character_partners WHERE citizenid = ? OR partner = ?', { citizenid, citizenid })
    MySQL.query.await('DELETE FROM character_invites WHERE citizenid = ? OR target = ?', { citizenid, citizenid })

    return true, 'Karakter dihapus'
end)

lib.callback.register('sm_characters:spawn', function(source, id)
    local player = QBCore.Functions.GetPlayer(source)
    if not player then return false, 'Karakter belum dimuat' end

    local position = player.PlayerData.position
    if not position or not position.x then position = locationById(id) end
    if not position or not position.x then position = DEFAULT_SPAWN end

    local heading = position.w or position.a or 0.0

    SetEntityCoords(source, position.x + 0.0, position.y + 0.0, position.z + 0.0, false, false, false, false)
    SetEntityHeading(source, heading)
    SetPlayerRoutingBucket(source, 0)

    return true
end)

-- ----------------------------------------------------------------------------
-- Partner
-- ----------------------------------------------------------------------------

local function myCitizen(source)
    local player = QBCore.Functions.GetPlayer(source)
    return player and player.PlayerData.citizenid or nil
end

local function partnerOf(citizenid)
    return MySQL.scalar.await('SELECT partner FROM character_partners WHERE citizenid = ?', { citizenid })
end

local function pushPartnerState(source)
    local citizenid = myCitizen(source)
    if not citizenid then return end

    local partner = partnerOf(citizenid)
    local payload = { partner = nil }

    if partner then
        local info = MySQL.scalar.await('SELECT charinfo FROM players WHERE citizenid = ?', { partner })
        payload.partner = { citizenid = partner, name = fullName(decode(info)) }
    end

    local outgoing = MySQL.query.await('SELECT target AS citizenid FROM character_invites WHERE citizenid = ?', { citizenid }) or {}
    local incoming = MySQL.query.await('SELECT citizenid FROM character_invites WHERE target = ?', { citizenid }) or {}

    payload.outgoing = outgoing
    payload.incoming = incoming

    TriggerClientEvent('sm_characters:client:partnerSync', source, payload)
end

lib.callback.register('sm_characters:partner:invite', function(source, target)
    local citizenid = myCitizen(source)
    if not citizenid then return false, 'Karakter belum dimuat' end
    if type(target) ~= 'string' then return false, 'Target tidak valid' end
    if target == citizenid then return false, 'Tidak bisa mengundang diri sendiri' end

    if partnerOf(citizenid) then return false, 'Anda sudah punya partner' end
    if partnerOf(target) then return false, 'Target sudah punya partner' end

    local targetLicense = MySQL.scalar.await('SELECT license FROM players WHERE citizenid = ?', { target })
    if not targetLicense then return false, 'Karakter target tidak ditemukan' end

    local targetSrc
    for src, player in pairs(QBCore.Players) do
        if player.PlayerData.citizenid == target then targetSrc = src break end
    end
    if not targetSrc then return false, 'Target sedang offline' end

    local exists = MySQL.scalar.await('SELECT citizenid FROM character_invites WHERE citizenid = ? AND target = ?', { citizenid, target })
    if exists then return false, 'Undangan sudah dikirim' end

    MySQL.insert.await('INSERT INTO character_invites (citizenid, target, expires) VALUES (?, ?, ?)',
        { citizenid, target, os.time() + INVITE_TTL })

    notify(targetSrc, 'Undangan partner masuk', 'inform')
    pushPartnerState(targetSrc)
    pushPartnerState(source)

    return true, 'Undangan dikirim'
end)

lib.callback.register('sm_characters:partner:respond', function(source, inviter, accept)
    local citizenid = myCitizen(source)
    if not citizenid then return false, 'Karakter belum dimuat' end

    local expires = MySQL.scalar.await('SELECT expires FROM character_invites WHERE citizenid = ? AND target = ?', { inviter, citizenid })
    if not expires then return false, 'Undangan tidak ditemukan' end

    MySQL.query.await('DELETE FROM character_invites WHERE citizenid = ? AND target = ?', { inviter, citizenid })

    if not accept then
        pushPartnerState(source)
        return true, 'Undangan ditolak'
    end

    if expires < os.time() then return false, 'Undangan kedaluwarsa' end
    if partnerOf(citizenid) or partnerOf(inviter) then return false, 'Salah satu sudah punya partner' end

    local now = os.time()
    MySQL.query.await('DELETE FROM character_partners WHERE citizenid IN (?, ?)', { citizenid, inviter })
    MySQL.insert.await('INSERT INTO character_partners (citizenid, partner, since) VALUES (?, ?, ?), (?, ?, ?)',
        { citizenid, inviter, now, inviter, citizenid, now })

    local inviterSrc
    for src, player in pairs(QBCore.Players) do
        if player.PlayerData.citizenid == inviter then inviterSrc = src break end
    end

    if inviterSrc then
        notify(inviterSrc, 'Undangan partner diterima', 'success')
        pushPartnerState(inviterSrc)
    end

    pushPartnerState(source)
    return true, 'Undangan diterima'
end)

lib.callback.register('sm_characters:partner:cancel', function(source, target)
    local citizenid = myCitizen(source)
    if not citizenid then return false, 'Karakter belum dimuat' end

    MySQL.query.await('DELETE FROM character_invites WHERE citizenid = ? AND target = ?', { citizenid, target })
    pushPartnerState(source)

    return true, 'Undangan dibatalkan'
end)

lib.callback.register('sm_characters:partner:remove', function(source, target)
    local citizenid = myCitizen(source)
    if not citizenid then return false, 'Karakter belum dimuat' end

    local partner = partnerOf(citizenid)
    if not partner then return false, 'Tidak ada partner' end
    if target and target ~= partner then return false, 'Bukan partner Anda' end

    MySQL.query.await('DELETE FROM character_partners WHERE citizenid IN (?, ?)', { citizenid, partner })
    MySQL.query.await('DELETE FROM character_invites WHERE citizenid IN (?, ?) OR target IN (?, ?)', { citizenid, partner, citizenid, partner })

    pushPartnerState(source)

    for src, player in pairs(QBCore.Players) do
        if player.PlayerData.citizenid == partner then
            notify(src, 'Partner Anda memutuskan hubungan', 'inform')
            pushPartnerState(src)
            break
        end
    end

    return true, 'Partner diputus'
end)
