local garage = {}

function garage.prepareVehicles()
    local vehicles = lib.cblt.await(
        'rhd_garage:server:fetchGarageVehicles',
        500
    )

    if not vehicles or type(vehicles) ~= 'table' then return {} end

    local zoneId = PlayerState?.rhd_garage_zoneId
    local garage_data = ClientLocations.getGarages(zoneId)?.garageData
    if not garage_data then return {} end

    ---@cast garage_data GarageData

    local filtered = {}
    local filterIndex = 1

    for i=1, #vehicles do
        local vehicle = vehicles[i]
        local model = vehicle.model

        if not vehicle.name or vehicle.name == 'Unknown' or tonumber(vehicle.name) then
            local hash = type(model) == 'string' and joaat(model) or model
            local displayName = GetDisplayNameFromVehicleModel(hash)
            local label = GetLabelText(displayName)

            if label and label ~= 'NULL' then
                vehicle.name = label
            elseif displayName and displayName ~= 'CARNOTFOUND' then
                vehicle.name = displayName
            end
        end

        vehicle.class = GetVehicleClassFromName(model)

        local garage_class = garage_data.vehicle_class or 'all'
        local validClass = utils.convertClass(garage_class)

        if lib.array.includes(validClass, vehicle.class) then
            filtered[filterIndex] = vehicle
            filterIndex = filterIndex + 1
        end
    end

    return filtered
end

---@param garageId GarageId
---@param garageData GarageData
function garage.openGarage(garageId, garageData)
    SendNUIMessage({
        action = 'openGarage',
        data = {
            name = garageId,
            garageIcon = garageData.garageIcon or 'warehouse',
        }
    })
    SetNuiFocus(true, true)
end

function garage.storeVehicle(self, vehicle, isAutoStore)
    if self.garageData.category == 'depot' then
        self.busy = false return
    end

    if not vehicle or vehicle == 0 or not DoesEntityExist(vehicle) then
        self.busy = false return
    end

    local class = GetVehicleClass(vehicle)
    local garage_class = self.garageData.vehicle_class or 'all'
    local validClass = utils.convertClass(garage_class)

    if not lib.array.includes(validClass, class) then
        self.busy = false
        return sm_core:Notify('This vehicle cannot be stored in this garage', 'error')
    end

    local state = Entity(vehicle).state
    local props = lib.getVehicleProperties(vehicle)

    if not props then
        self.busy = false
        return sm_core:Notify('Gagal membaca data kendaraan, coba lagi', 'error')
    end

    props.bodyHealth = GetVehicleBodyHealth(vehicle)
    props.engineHealth = GetVehicleEngineHealth(vehicle)
    props.fuelLevel = exports.LegacyFuel:GetFuel(vehicle)
    props.harnessDurability = state.harnessDurability

    state:set('rhd_garage_lockvehicle', true, true)
    state:set('rhd_garage_vehicleProps', props, true)
    state:set('rhd_garage_zoneId', self.garageId, true)

    if GetPedInVehicleSeat(vehicle, -1) == cache.ped then
        TaskLeaveVehicle(cache.ped, vehicle, 1)

        local timeout = GetGameTimer() + 5000
        repeat
            Wait(250)
        until not cache.vehicle or GetGameTimer() > timeout
    end

    self.busy = false

    local netId = NetworkGetNetworkIdFromEntity(vehicle)
    TriggerTokenEvent('rhd_garage:server:store_vehicle', netId, isAutoStore, self.garageId, props)
end

function garage.toggleFavorite(plate)
    local KVP = 'rhd_garage:favorites'
    local favorites = utils.kvpGet(KVP) or {}

    if favorites[plate] then
        favorites[plate] = nil
    else
        favorites[plate] = true
    end

    utils.kvpSet(KVP, favorites)
    return favorites
end

function garage.getFavorites()
    local KVP = 'rhd_garage:favorites'
    return utils.kvpGet(KVP) or {}
end

lib.callback.register('rhd_garage:client:spawn_vehicle', function (data)
    local mods = data.mods
    local coords = data.coords

    local vehicle, err = utils.spawnCar(mods.model, coords, true)

    if not vehicle then
        local errorMsg = err or 'Failed to spawn vehicle'
        return sm_core:Notify(errorMsg, 'error')
    end

    lib.setVehicleProperties(vehicle, mods)

    if mods.harnessDurability and mods.harnessDurability > 0 then
        Entity(vehicle).state:set('harnessDurability', mods.harnessDurability, true)
    end

    CreateThread(function ()
        Wait(500)
        SetVehicleBodyHealth(vehicle --[[@as integer]], mods.bodyHealth)
        SetVehicleEngineHealth(vehicle --[[@as integer]], mods.engineHealth)
        exports.LegacyFuel:SetFuel(vehicle, mods.fuelLevel or 100)

        if mods.doors then
            for i = 1, #mods.doors do
                SetVehicleDoorBroken(vehicle, mods.doors[i], true)
            end
        end

        if mods.windows then
            for i = 1, #mods.windows do
                RemoveVehicleWindow(vehicle, mods.windows[i])
            end
        end

        if mods.tyres then
            for tyre, state in pairs(mods.tyres) do
                SetVehicleTyreBurst(vehicle, tonumber(tyre) --[[@as number]], state == 2, 1000.0)
            end
        end
    end)

    TaskWarpPedIntoVehicle(cache.ped, vehicle --[[@as integer]], -1)
    TriggerEvent('sm_vehiclekeys:client:setVehicleOwner', GetVehicleNumberPlateText(vehicle --[[@as integer]]))

    return NetworkGetNetworkIdFromEntity(vehicle --[[@as integer]])
end)

local lastVehicle = 0
local lastVehiclePlate = nil
local lastVehicleProps = nil

CreateThread(function()
    while true do
        local sleep = 2000
        if cache.vehicle and DoesEntityExist(cache.vehicle) then
            -- Cache the plate and damage properties while inside the vehicle
            -- in case it gets unexpectedly deleted (e.g. /dv)
            local plateText = GetVehicleNumberPlateText(cache.vehicle)
            if plateText then
                lastVehiclePlate = string.gsub(plateText, '^%s*(.-)%s*$', '%1')
                local fullProps = lib.getVehicleProperties(cache.vehicle)
                fullProps.bodyHealth = GetVehicleBodyHealth(cache.vehicle)
                fullProps.engineHealth = GetVehicleEngineHealth(cache.vehicle)
                fullProps.fuelLevel = exports.LegacyFuel:GetFuel(cache.vehicle)
                fullProps.harnessDurability = Entity(cache.vehicle).state.harnessDurability
                lastVehicleProps = fullProps
            end
        else
            sleep = 1000
        end
        Wait(sleep)
    end
end)

lib.onCache('vehicle', function(vehicle)
    if vehicle then
        -- Player entered a vehicle
        lastVehicle = vehicle
    elseif lastVehicle ~= 0 then
        -- Player left the vehicle or vehicle was deleted
        local veh = lastVehicle
        lastVehicle = 0

        local plate = lastVehiclePlate
        local props = nil
        
        if DoesEntityExist(veh) then
            local plateText = GetVehicleNumberPlateText(veh)
            plate = plateText and string.gsub(plateText, '^%s*(.-)%s*$', '%1') or ''
            
            props = lib.getVehicleProperties(veh)
            props.bodyHealth = GetVehicleBodyHealth(veh)
            props.engineHealth = GetVehicleEngineHealth(veh)
            props.fuelLevel = exports.LegacyFuel:GetFuel(veh)
            props.harnessDurability = Entity(veh).state.harnessDurability
        else
            props = lastVehicleProps
        end

        if plate and plate ~= '' and props then
            TriggerServerEvent('rhd_garage:server:update_vehicle_props', plate, props)
        end
        
        lastVehiclePlate = nil
        lastVehicleProps = nil
    end
end)

local instances = {}

local garage_config = require 'config.garage'
local garage_auto_store_config = require 'data.garage_auto_store'

local function infoThread(self)

    if self.garageData then
        local garageCoords = self.coords
        local playerCoords = GetEntityCoords(cache.ped)
        local maxZDiff = self.garageData.locations.maxZDiff or 3.0

        if math.abs(playerCoords.z - garageCoords.z) > maxZDiff then
            if self.infoUi then
                infoUi.remove('rhd_garage:info')
                self.infoUi = nil
            end
            return
        end
    end

    local description
    local inVeh = cache.vehicle

    if self.isAutoStore then
        description = [[
Zona ini akan otomatis menyimpan kendaraan Anda saat
memasuki area dengan kendaraan.
        ]]
    else
        description = inVeh and not (self.garageData.category == 'depot') and
            'Tekan [kbd]E[/kbd] untuk simpan kendaraan' or not inVeh and
            'Tekan [kbd]E[/kbd] untuk akses garasi'
    end

    if not self.isAutoStore and utils.isDead() then
        description = nil
    end

    if not description or not self.canAccess then
        if self.infoUi then
            infoUi.remove('rhd_garage:info')
            self.infoUi = nil
        end
        return
    end

    if not self.infoUi or self.infoUi ~= description then
        infoUi.show({
            id = 'rhd_garage:info',
            title = self.garageId,
            icon = 'fa-solid fa-warehouse',
            description = description,
        })
        self.infoUi = description
    end
end

local function interactThread(self)
    if not self.canAccess then return end

    DisableControlAction(0, 38, true)

    if not self.busy and IsDisabledControlJustPressed(0, 38) then
        if utils.isDead() then
            return sm_core:Notify('Kamu tidak bisa melakukan ini saat mati', 'error')
        end

        local garageCoords = self.coords
        local playerCoords = GetEntityCoords(cache.ped)
        local maxZDiff = self.garageData.locations.maxZDiff or 3.0

        if math.abs(playerCoords.z - garageCoords.z) > maxZDiff then
            return
        end

        local vehicle = cache.vehicle
        if vehicle and not DoesEntityExist(vehicle) then
            vehicle = nil
        end
        self.busy = true

        if vehicle then
            Client.storeVehicle(self, vehicle)
        else
            self.busy = false
            Client.openGarage(self.garageId, self.garageData)
        end
    end
end

local function onEnter(self)
    self.infoUi = nil

    PlayerState:set('rhd_garage_zoneId', self.garageId, true)
    self.canAccess = utils.canAccess(self.garageData.access)

    if self.garageData.locations then
        local garageCoords = self.coords
        local playerCoords = GetEntityCoords(cache.ped)
        local maxZDiff = self.garageData.locations.maxZDiff or 3.0

        if math.abs(playerCoords.z - garageCoords.z) > maxZDiff then
            self.canAccess = false
        end
    end

    if cache.vehicle and self.garageData.autoStoreVehicle and self.canAccess then
        self.busy = true
        Client.storeVehicle(self, cache.vehicle, true)
    end
end

local function onExit(self)
    self.infoUi = nil
    infoUi.remove('rhd_garage:info')
    PlayerState:set('rhd_garage_zoneId', nil, true)
end

local function onRemove(self)
    if self.infoUi then
        infoUi.remove('rhd_garage:info')
        self.infoUi = nil
    end
    if self.blipId then
        RemoveBlip(self.blipId)
    end
    PlayerState:set('rhd_garage_zoneId', nil, true)
end

local function inside(self)
    if not self.garageData.hideInfoUi then
        self:infoThread()
    end
    self:interactThread()
end

---@param garageId GarageId
---@param garageData GarageData
local function prepareInstance(garageId, garageData)
    local instance
    local ltype = garageData.locations.type

    if ltype == 'zones' then
        instance = lib.zones.poly(garageData.locations.polygon)
    elseif ltype == 'points' then
        local coords = garageData.locations.coords
        local distance = garageData.locations.distance or 5.0
        instance = lib.points.new(coords, distance)
    end

    instance.garageId = garageId --[[@as GarageId]]
    instance.garageData = garageData --[[@as GarageData]]

    if garageData.blipType then
        local blipData = garage_config.blip_data[garageData.blipType]
        instance.blipId = utils.createGarageBlip(instance.coords, garageId, blipData, true)
    end

    function instance:updateAccess()
        self.canAccess = utils.canAccess(self.garageData.access)
    end

    instance.infoThread = infoThread
    instance.interactThread = interactThread
    instance.onEnter = onEnter
    instance.onExit = onExit

    if ltype == 'zones' then
        instance.inside = inside
    elseif ltype == 'points' then
        instance.nearby = inside
    end

    instance.onRemove = onRemove

    return instance
end

---@param garageId GarageId
---@param polygon GaragePolygon
local function prepareAutoStoreInstance(garageId, polygon)
    local instance = lib.zones.poly(polygon)

    instance.garageId = garageId --[[@as GarageId]]
    instance.isAutoStore = true

    function instance:onEnter()
        self.infoUi = nil
        if not cache.vehicle or self.busy then return end

        self.busy = true
        local coreInstance = instances[garageId]?.core

        if not coreInstance then
            return lib.print.warn(('Garage instance not found for auto store [garageId: %s]'):format(garageId))
        end

        local canAccess = utils.canAccess(coreInstance.garageData.access)
        if not canAccess then return end

        Client.storeVehicle(coreInstance, cache.vehicle, true)
    end

    function instance:inside()
        infoThread(self)
    end
    
    function instance:onExit()
        self.busy = false
        self.infoUi = nil
        infoUi.remove('rhd_garage:info')
    end

    return instance
end

---@param garageId GarageId
---@param data GaragePolygon | GaragePolygon[]
local function prepareAutoStoreInstances(garageId, data)
    if data.points then
        return { prepareAutoStoreInstance(garageId, data) }
    end

    local zones = {}

    for i = 1, #data do
        zones[i] = prepareAutoStoreInstance(garageId, data[i])
    end

    return zones
end

local function removeAutoStoreZones(zones)
    for i = 1, #zones do
        zones[i]:remove()
    end
end

local function checkInstance(garageId, type, data)
    if not instances[garageId] then
        instances[garageId] = {}
    end

    local fn = type == 'core' and prepareInstance or
                 type == 'autoStore' and prepareAutoStoreInstances

    if fn and not instances[garageId][type] then
        instances[garageId][type] = fn(garageId, data)
    end
end

---@param garages GarageConfig
local function prepareGarageLocations(garages)
    for garageId, garageData in pairs(garages) do
        checkInstance(garageId, 'core', garageData)

        local autoStoreConfig = garage_auto_store_config[garageId]
        if autoStoreConfig then
            checkInstance(garageId, 'autoStore', autoStoreConfig)
        end
    end
end

RegisterNetEvent('rhd_garage:client:loadGarageLocations', function (garages)
    if source == '' then return end
    prepareGarageLocations(garages)
end)

RegisterNetEvent('rhd_garage:client:addGarageLocation', function (garageId, garageData)
    if source == '' then return end
    prepareGarageLocations({ [garageId] = garageData })
end)

RegisterNetEvent('rhd_garage:client:addAutoStoreLocation', function (garageId, polygon)
    if source == '' then return end

    if garage_auto_store_config[garageId] then
        return lib.print.warn(('Auto store configuration already exists for garageId %s, skipping add'):format(garageId))
    end
    
    garage_auto_store_config[garageId] = polygon
    checkInstance(garageId, 'autoStore', polygon)
end)

RegisterNetEvent('rhd_garage:client:removeAutoStoreLocation', function (garageId)
    if source == '' then return end

    garage_auto_store_config[garageId] = nil

    local entry = instances[garageId]
    if not entry or not entry.autoStore then
        return lib.print.warn(('Auto store instance not found for garageId %s, skipping remove'):format(garageId))
    end

    removeAutoStoreZones(entry.autoStore)
    instances[garageId].autoStore = nil
end)

RegisterNetEvent('rhd_garage:client:removeGarageLocation', function (garageId)
    if source == '' then return end

    local entry = instances[garageId]
    if not entry then return end

    if entry.autoStore then
        removeAutoStoreZones(entry.autoStore)
    end

    if entry.core then
        if entry.core.onRemove then
            entry.core:onRemove()
        end

        entry.core:remove()
    end

    instances[garageId] = nil
end)

RegisterNetEvent('rhd_garage:client:autoStoreVehicleByPlate', function (plate, garageId)
    if source == '' then return end

    local entry = instances[garageId]
    if not entry or not entry.core then
        return lib.print.warn(('Auto store instance not found for garageId %s, cannot auto store vehicle [plate: %s]'):format(garageId, plate))
    end

    local vehicle = utils.findPlateOnClient(plate)

    if not (vehicle and DoesEntityExist(vehicle)) then
        return lib.print.warn(('Vehicle with plate %s not found for auto store in garageId %s'):format(plate, garageId))
    end

    Client.storeVehicle(entry.core, vehicle, true)
end)

RegisterNetEvent('QBCore:Client:OnJobUpdate', function (job)
    if source == '' then return end

    local zoneId = PlayerState.rhd_garage_zoneId
    if not zoneId then return end

    local entry = instances[zoneId]
    if not entry or not entry.core then return end

    entry.core:updateAccess()
end)

exports('GetGarages', function ()
    local out = {}
    for garageId, entry in pairs(instances) do
        out[garageId] = entry.core
    end
    return out
end)

local storage = {}

local query = {
    fetchPlayerVehicles = {
        shared = {
            query = [[
                SELECT 
                    pv.vehicle, pv.hash, pv.plate, pv.state, pv.depotprice,
                    JSON_EXTRACT(pv.mods, '$.plateIndex') AS plateIndex,
                    CONCAT(
                        JSON_UNQUOTE(JSON_EXTRACT(p.charinfo, '$.firstname')),
                        ' ',
                        JSON_UNQUOTE(JSON_EXTRACT(p.charinfo, '$.lastname'))
                    ) AS owner_name
                FROM player_vehicles pv
                JOIN players p ON pv.citizenid = p.citizenid
                WHERE pv.garage = ? AND pv.state = ?
            ]],
            values = function (data)
                return { data.garageId, 1 }
            end
        },
        default = {
            query = [[
                SELECT vehicle, hash, plate, state, depotprice, JSON_EXTRACT(mods, '$.plateIndex') AS plateIndex FROM player_vehicles
                WHERE citizenid = ? AND garage = ? AND state = ?
            ]],
            values = function (data)
                return { data.citizenid, data.garageId, 1 }
            end
        },
        depot = {
            query = [[
                SELECT 
                    pv.vehicle, pv.hash, pv.plate, pv.state, pv.depotprice, JSON_EXTRACT(pv.mods, '$.plateIndex') AS plateIndex, (
                        SELECT 1 FROM rhd_pimpound rp WHERE pv.citizenid = rp.citizenid AND pv.plate = rp.plate) AS impounded
                FROM player_vehicles pv
                WHERE pv.citizenid = ? AND (pv.state = 0 OR pv.state = 2)
            ]],
            values = function (data)
                return { data.citizenid }
            end
        }
    },

---@param citizenid string
---@param plate string
---@param garageId GarageId
---@param request table
---@return table|nil
function storage.getVehicleData(citizenid, plate, garageId, request)
    local garage_data = glocations.registered[garageId] ---@type GarageData
    if not garage_data then return nil end

    if not request or #request == 0 then return nil end

    local select_clause = table.concat(request, ', ')
    local q, values

    if garage_data.category == 'shared' then
        q = string.format('SELECT %s FROM player_vehicles WHERE plate = ?', select_clause)
        values = { plate }
    else
        q = string.format('SELECT %s FROM player_vehicles WHERE citizenid = ? AND plate = ?', select_clause)
        values = { citizenid, plate }
    end

    local results = db:query_async(q, values)
    if not results or #results == 0 then return nil end

    local data = results[1]
    if data.mods then
        data.mods = json.decode(data.mods)

        if type(data.mods.model) == 'string' then
            data.mods.model = joaat(data.mods.model)
        end

    end

    return data
end

---@param citizenid string
---@param plate string
---@param garageId GarageId
---@param newData table
---@return boolean Success
function storage.setVehicleData(citizenid, plate, garageId, newData)
    local garage_data = glocations.registered[garageId] ---@type GarageData
    if not garage_data then return false end

    if not newData or next(newData) == nil then return false end

    local set_parts = {}
    local values = {}

    for k, v in pairs(newData) do
        set_parts[#set_parts + 1] = k .. ' = ?'
        if k == 'mods' and type(v) == 'table' then
            values[#values + 1] = json.encode(v)
        else
            values[#values + 1] = v
        end
    end

    local q
    if garage_data.category == 'shared' then
        q = string.format('UPDATE player_vehicles SET %s WHERE plate = ?', table.concat(set_parts, ', '))
        values[#values + 1] = plate
    else
        q = string.format('UPDATE player_vehicles SET %s WHERE citizenid = ? AND plate = ?', table.concat(set_parts, ', '))
        values[#values + 1] = citizenid
        values[#values + 1] = plate
    end

    return db:update_async(q, values) > 0
end

---@param hash number|string|nil
---@return integer|nil
local function normalizeHash(hash)
    if type(hash) == 'string' and not tonumber(hash) then
        hash = joaat(hash)
    end
    hash = tonumber(hash)
    hash = hash and math.tointeger(hash)
    if not hash or hash == 0 then return nil end
    return hash & 0xFFFFFFFF
end

---@param row {vehicle?: string, hash?: number|string}
---@return integer|nil
function storage.rowModel(row)
    local name = row.vehicle
    if type(name) == 'string' and name ~= '' then
        return normalizeHash(name)
    end
    return normalizeHash(row.hash)
end

---@param a number|string|nil
---@param b number|string|nil
---@return boolean
function storage.sameModel(a, b)
    local na = normalizeHash(a)
    return na ~= nil and na == normalizeHash(b)
end

---@param citizenid string
---@param vehicle {model: string|number, plate: string}
---@param garageId GarageId
---@return boolean isOwner
function storage.checkVehicleOwner(citizenid, vehicle, garageId)
    local garage_data = glocations.registered[garageId] ---@type GarageData
    if not garage_data then return false end

    local vehicleData = storage.getVehicleData(citizenid, vehicle.plate, garageId, { 'mods', 'vehicle', 'hash' })
    if not vehicleData then return false end

    local rowModel = storage.rowModel(vehicleData)
    if rowModel then
        return storage.sameModel(rowModel, vehicle.model)
    end

    return storage.sameModel(vehicleData.mods?.model, vehicle.model)
end

---@param plate string
---@return string|nil citizenid
function storage.getVehicleOwnerByPlate(plate)
    local results = db:query_async('SELECT citizenid FROM player_vehicles WHERE plate = ?', { plate })
    if not results or #results == 0 then return nil end

    if #results > 1 then
        lib.print.warn(('Plate %s dimiliki %d kendaraan, pemilik tidak dapat ditentukan'):format(plate, #results))
        return nil
    end

    return results[1].citizenid
end

---@param citizenid string
---@param garageId GarageId
---@return table PlayerVehicles[]
function storage.fetchGarageVehicles(citizenid, garageId)
    local garage_data = glocations.registered[garageId] ---@type GarageData
    if not garage_data then return {} end

    local q = query.fetchPlayerVehicles[garage_data.category]
    local results = db:query_async(q.query, q.values({ citizenid = citizenid, garageId = garageId }))

    local vehicles = {}
    local isDepot = garage_data.category == 'depot'
    local isShared = garage_data.category == 'shared'
    local isDefault = garage_data.category == 'default'

    for i=1, #results do
        local v = results[i]
        local outside = utils.findPlateOnServer(v.plate)
        local vehdata = qb_vehicles.hash(v.hash) or qb_vehicles(v.vehicle)
        local vehmodel = vehdata?.model or (v.vehicle or v.hash)

        local metadata = {
            plateIndex = v.plateIndex,
        }

        if isDepot then
            metadata.depotPrice = utils.calculateDepotPrice(
                vehmodel,
                v.depotprice
            )
            metadata.outside = outside
            metadata.impounded = v.state == 2 and v.impounded
        elseif isShared then
            metadata.owner_name = v.owner_name
        end

        local vehicle = {
            name = vehdata?.name or 'Unknown',
            model = vehmodel,
            plate = v.plate,
            metadata = metadata
        }

        if (isDefault or isShared) and outside then
            goto continue
        end

        vehicles[#vehicles+1] = vehicle

        ::continue::
    end

    return vehicles
end

return storage

---@param gclass GarageVehicleClass | GarageVehicleClass[]
---@return number[] VehicleClass
function utils.convertClass(gclass)
    local class = {}
    local typec = type(gclass)

    if typec == 'string' and garage_class[gclass] then
        class = lib.array.merge(class, garage_class[gclass])
    elseif typec == 'table' and gclass[1] then
        for i=1, #gclass do
            local gc = gclass[i]
            if garage_class[gc] then
                class = lib.array.merge(class, garage_class[gc])
            end
        end
    end
    return class
end

function utils.kvpGet(key)
    local raw = GetResourceKvpString(key)
    if not raw or raw == '' then return nil end
    local ok, val = pcall(json.decode, raw)
    return ok and val or nil
end

---@param key  string
---@param data table
function utils.kvpSet(key, data)
    local ok, encoded = pcall(json.encode, data)
    if ok then SetResourceKvp(key, encoded) end
end

    ---@param val number
    ---@return number? Value
    local function prefixValue(val)
        if type(val) ~= 'number' then return end
        return math.min(100, math.max(0, math.floor(val)))
    end

    ---@param vehicle integer
    ---@return table VehiclePerformance
    local function getPerformanceStats(vehicle)
        local fInitialDriveMaxFlatVel = GetVehicleHandlingFloat(vehicle, "CHandlingData", "fInitialDriveMaxFlatVel")
        local fInitialDriveForce = GetVehicleHandlingFloat(vehicle, "CHandlingData", "fInitialDriveForce")
        local fBrakeForce = GetVehicleHandlingFloat(vehicle, "CHandlingData", "fBrakeForce")
        local fTractionCurveMax = GetVehicleHandlingFloat(vehicle, "CHandlingData", "fTractionCurveMax")
        local fTractionCurveMin = GetVehicleHandlingFloat(vehicle, "CHandlingData", "fTractionCurveMin")
        local fSteeringLock = GetVehicleHandlingFloat(vehicle, "CHandlingData", "fSteeringLock")
        local fSuspensionUpperLimit = GetVehicleHandlingFloat(vehicle, "CHandlingData", "fSuspensionUpperLimit")
        local fSuspensionLowerLimit = GetVehicleHandlingFloat(vehicle, "CHandlingData", "fSuspensionLowerLimit")

        local maxSpeedKMH = (fInitialDriveMaxFlatVel * 0.75) * 3.6
        local suspensionTravel = fSuspensionUpperLimit - fSuspensionLowerLimit

        local stats = {
            speed = prefixValue((maxSpeedKMH / 530.0) * 100),
            acceleration = prefixValue((fInitialDriveForce / 1.3) * 100),
            braking = prefixValue((fBrakeForce / 2.0) * 100),
            handling = prefixValue(((fSteeringLock / 42.0) * 0.7 + ((fTractionCurveMax + fTractionCurveMin) / 5.5) * 0.3) * 100),
            offroad = prefixValue((suspensionTravel / 0.35) * 100),
            drift = prefixValue((1.0 - (fTractionCurveMin / math.max(fTractionCurveMax, 0.001))) * 100),
        }

        return stats
    end

    ---@param data {plate: string, model: string|integer}
    ---@return table? VehicleStats
    function utils.getVehicleStats(data)
        local model = data.model
        local plate = data.plate

        local coords = GetEntityCoords(cache.ped)
        local heading = GetEntityHeading(cache.ped)
        
        if not pcall(lib.requestModel, model) then
            return nil
        end

        local tempIndex = #tempVehicle + 1
        local vehEntity = CreateVehicle(model, coords.x, coords.y, coords.z, heading, true, false)
        FreezeEntityPosition(vehEntity, true)
        SetEntityAlpha(vehEntity, 25, false)
        SetEntityCollision(vehEntity, false, false)
        SetVehicleNumberPlateText(vehEntity, plate)

        tempVehicle[tempIndex] = vehEntity
        local performance = getPerformanceStats(vehEntity)
        local server_data = lib.cblt.await(
            'rhd_garage:server:fetchServerVehicleDetails', false, plate
        )

        if not performance then
            return
        end

        performance.fuel = server_data.fuel
        performance.engine = server_data.engine
        performance.body = server_data.body

        if performance.fuel > 100 then performance.fuel = 100 end
        if performance.engine > 100 then performance.engine = 100 end
        if performance.body > 100 then performance.body = 100 end

        local vehdata = qb_vehicles.hash(model) or qb_vehicles(model --[[@as string]])

        local details = {
            name = vehdata.name,
            model = model,
            plate = plate,
            stats = performance,
            trunkItems = lib.array.filter(server_data.trunkItems, function (val)
                return val
            end),
            gloveboxItems = lib.array.filter(server_data.gloveboxItems, function (val)
                return val
            end),
        }

        DeleteEntity(vehEntity)

        local attempt = 0
        repeat
            pcall(DeleteEntity, vehEntity)
            Wait(500)
            attempt = attempt + 1
        until not DoesEntityExist(vehEntity) or attempt >= 5

        return details
    end

    
    ---@param model integer | string
    ---@param coords table | vector3
    ---@return number|boolean? vehicle
    ---@return string? error
    function utils.spawnCar(model, coords, networked)
        model = type(model) == "number" and model or joaat(model)

        local vector = vec(coords.x, coords.y, coords.z, coords.w or 0.0)
        local isNetworked = networked == nil or networked

        local playerCoords = GetEntityCoords(cache.ped)
        if not vector or not playerCoords then
            return false, 'Invalid coordinates'
        end

        local dist = #(playerCoords - vector.xyz)

        if dist > 424 then -- Onesync infinity Range (https://docs.fivem.net/docs/scripting-reference/onesync/)
            return false, 'Out of range'
        end

        local promise = promise.new()

        local function results(res)
            if promise then
                promise:resolve(res)
            end
        end

        CreateThread(function()
            if not pcall(lib.requestModel, model, 15000) then
                return results(nil)
            end
            
            local vehicle = CreateVehicle(model, vector.x, vector.y, vector.z, vector.w, isNetworked, true)

            if networked then
                local id = NetworkGetNetworkIdFromEntity(vehicle)
                SetNetworkIdCanMigrate(id, true)
                SetEntityAsMissionEntity(vehicle, true, true)
            end
            
            SetVehicleHasBeenOwnedByPlayer(vehicle, true)
            SetVehicleNeedsToBeHotwired(vehicle, false)
            SetModelAsNoLongerNeeded(model)
            SetVehRadioStation(vehicle, "OFF")

            RequestCollisionAtCoord(vector.x, vector.y, vector.z)
            while not HasCollisionLoadedAroundEntity(vehicle) do
                Wait(0)
            end

            results(vehicle)
        end)

        if promise then
            local vehicle = Citizen.Await(promise)
            
            if not vehicle or not DoesEntityExist(vehicle) then
                return false, 'Failed to load vehicle model'
            end

            return vehicle
        end
    end

    ---@param coords vector3
    ---@param blipData BlipData
    ---@param category boolean
    function utils.createGarageBlip(coords, garage, blipData, category)
        local blip = AddBlipForCoord(coords.x, coords.y, coords.z)
        SetBlipSprite(blip, blipData?.sprite or 285)
        SetBlipScale(blip, 0.8)
        SetBlipColour(blip, blipData?.color or 3)
        SetBlipAsShortRange(blip, true)
        BeginTextCommandSetBlipName("STRING")
        AddTextComponentString(garage)
        EndTextCommandSetBlipName(blip)

        if category then
            SetBlipCategory(blip, 10)
        end
    end

    ---@param jobs GarageGroupAccess
    function utils.checkJobs(playerJob, jobs)
        if not playerJob then return false end

        local jn = playerJob.name
        local jt = playerJob.type
        local jg = playerJob.grade.level


        local tt = type(jobs)

        if tt == 'string' then
            return jn == jobs or jt == jobs
        end

        if tt == 'table' then
            if lib.array.isArray(jobs) then
                return lib.table.contains(jobs, jn) or lib.table.contains(jobs, jt)
            end

            local requirement = jobs[jn] or jobs[jt]
            if not requirement then return false end

            local rt = type(requirement)

            if rt == 'number' then
                return jg >= requirement
            end

            if rt == 'table' and lib.array.isArray(requirement) then
                return lib.table.contains(requirement, jg)
            end

            return false
        end

        return false
    end

    ---@param check GarageAccessBy
    function utils.canAccess(check)
        if not check then
            return true
        end

        local jobs = check.groups
        local citizenids = check.citizenids

        ---@cast citizenids -nil

        local job = QBPlayerData.job or sm_core:GetPlayerData()?.job
        local citizenid = QBPlayerData.citizenid or sm_core:GetPlayerData()?.citizenid

        local tc = type(citizenids)
        local haveAccess = tc == 'string' and citizenid == citizenids

        if not haveAccess and tc == 'table' and lib.array.isArray(citizenids) then
            haveAccess = lib.array.includes(citizenids, citizenid)
        end

        if not haveAccess and jobs then
            haveAccess = utils.checkJobs(job, jobs)
        end

        return haveAccess
    end

    ---@return boolean
    function utils.isDead()
        local metadata = QBPlayerData.metadata or sm_core:GetPlayerData()?.metadata
        if metadata and (metadata.isdead or metadata.inlaststand) then
            return true
        end
        return IsEntityDead(cache.ped)
    end

    ---@param model string
    ---@param default number
    ---@return number
    function utils.calculateDepotPrice(model, default)
        local vehdata = qb_vehicles.hash(model) or qb_vehicles(model)
        local category = vehdata and vehdata.category

        local price = 0

        if type(default) == "number" and default >= 1 then
            price = default
        else
            price = depot_price[category] or 5000
        end

        return price
    end

    ---@param coords vector3
    ---@param heading number
    ---@param distance? number
    ---@return vector4? spawn_point
    function utils.getNearbySpawnPoints(coords, heading, distance)
        local angles = { 0, 90, 180, 270, 45, 135, 225, 315 }
        distance = distance or 3.0

        local spawn_point
        for _, angle in ipairs(angles) do
            local rad = math.rad(heading + angle)
            local x = coords.x + (math.sin(rad) * distance)
            local y = coords.y + (math.cos(rad) * distance)
            local z = coords.z

            local testCoords = vec(x, y, z, heading + angle)

            if not lib.getNearbyVehicles(vec3(x, y, z), distance, true)[1] then
                spawn_point = testCoords
                break
            end
        end
        return spawn_point
    end
