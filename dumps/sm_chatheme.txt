local Config = require "config.config"

local ipairs, pairs = ipairs, pairs
local GetGameTimer = GetGameTimer
local ceil = math.ceil

local Cooldowns = {}
local CooldownExpires = {}

for name, data in pairs(Config.Commands) do
    if data.cooldown and data.cooldown > 0 then
        Cooldowns[name] = data.cooldown * 1000
    end
end

---@param command string
---@return boolean
local function CheckCooldown(command)
    local duration = Cooldowns[command]
    if not duration then return true end

    local now = GetGameTimer()
    local expires = CooldownExpires[command]

    if expires and now < expires then
        lib.notify({
            title = 'Chat',
            description = ('Tunggu %d detik lagi untuk memakai /%s.'):format(ceil((expires - now) / 1000), command),
            type = 'error'
        })
        return false
    end

    CooldownExpires[command] = now + duration
    return true
end

local function BuildText(args)
    local text = ""
    for _, v in ipairs(args) do
        text = text .. " " .. v
    end
    return text
end

local function RegisterSimpleCommand(category, useMask)
    local command = Config.Commands[category].name

    RegisterCommand(command, function(source, args)
        local text = BuildText(args)
        if text:len() == 0 then return end
        if not CheckCooldown(command) then return end

        TriggerServerEvent('sm_chatheme:server:sendMessage', {
            category = category,
            message = text,
            isImage = false,
            mask = useMask and CheckMask() or false,
        })
    end, false)
end

local function GetWebLocales()
    local path = 'locales/' .. Config.Locale .. '.json'
    local file = LoadResourceFile(GetCurrentResourceName(), path)

    if not file then
        file = LoadResourceFile(GetCurrentResourceName(), 'locales/en.json')
    end

    if not file or file == '' then
        return {}
    end

    local success, content = pcall(json.decode, file)
    if not success or not content or not content['Web'] then
        return {}
    end

    return content['Web']
end

local function CheckNui()
    while not NuiLoaded do
        Wait(100)
    end
end

function SendVueMessage(action, payload)
    CheckNui()
    SendNUIMessage({
        action = action,
        payload = payload
    })
end

local function FetchNui()
    while not NuiLoaded do
        if NetworkIsSessionStarted() then
            SendNUIMessage({ action = "CHECK_NUI" })
        end
        Wait(2000)
    end

    SendVueMessage("SET_CONFIG", {
        Themes = Config.Themes,
        ThemeColors = Config.ThemeColors,
        MessageColors = Config.MessageColors,
        Categories = Config.Categories,
        QuickCommands = Config.QuickCommands,
        AllowedURLs = Config.ImageSettings.allowedURLs,
        Emojis = Config.Emojis,
        Locales = GetWebLocales(),
    })
end

function DrawDots(coords, text)
    local defaultScale = 0.65
    local camCoords = GetGameplayCamCoord()
    local dist = #(coords - camCoords)
    local scale = 200 / (GetGameplayCamFov() * dist)

    SetTextColour(255, 255, 255, 255)
    SetTextFont(0)
    SetTextScale(0.0, defaultScale * scale)
    SetTextDropshadow(0, 0, 0, 0, 55)
    SetTextDropShadow()
    SetTextCentre(true)

    BeginTextCommandDisplayText("STRING")
    AddTextComponentSubstringPlayerName(text)
    SetDrawOrigin(coords.x, coords.y, coords.z, 0)
    EndTextCommandDisplayText(0.0, 0.0)
    ClearDrawOrigin()
end

function DrawText3D(x, y, z, text, color)
    local camCoords = GetGameplayCamCoord()
    local dist = #(vector3(x, y, z) - camCoords)
    local scale = 200 / (GetGameplayCamFov() * dist)

    SetTextColour(color.r, color.g, color.b, color.a)
    SetTextScale(0.0, DrawTextScale * scale)
    SetTextFont(4)
    SetTextDropshadow(0, 0, 0, 0, 55)
    SetTextDropShadow()
    SetTextCentre(true)

    BeginTextCommandDisplayText("STRING")
    AddTextComponentSubstringPlayerName(text)
    SetDrawOrigin(x, y, z, 0)
    EndTextCommandDisplayText(0.0, 0.0)
    ClearDrawOrigin()
end

function UpdateRPTexts(source, data)
    for k, v in ipairs(RPTexts) do
        if v.source == source then
            RPTexts[k] = data
            return
        end
    end
    table.insert(RPTexts, data)
end

function CheckMask()
    if not Config.AnonymousWithMask then return false end

    local ped = cache.ped
    local pedModel = GetEntityModel(ped)

    if not Config.MaskIndexs[pedModel] then return false end

    local currentMaskIndex = GetPedDrawableVariation(ped, 1)

    if Config.AnyMask then
        return currentMaskIndex > 0
    end

    return Config.MaskIndexs[pedModel][tostring(currentMaskIndex)] == true
end

        RPTextsDUIThread()
    else
        RPTexts3DThread()

    Wait(5000)

    for k, v in ipairs(RPTexts) do

                SendVueMessage("REMOVE_3D_TEXT", { source = v.source })
            end
            table.remove(RPTexts, k)
            break
        end
    end
end)

RegisterNUICallback("SONG_PLAYING", function(data, cb)
    ActiveSong = {
        title = data.title or "Unknown Title",
        url = data.url or "",
        isPlaying = true,
        volume = Config.MusicSettings.defaultVolume,
    }

    if not ActiveSong.url or ActiveSong.url == "" then
        cb('ok')
        return
    end

    if GetResourceState("xsound") ~= "started" then
        cb('ok')
        return
    end

    exports.xsound:PlayUrl("sm_chatheme-song-" .. cache.serverId, ActiveSong.url, ActiveSong.volume, false)

    SendVueMessage("SONG_DATA", {
        title = ActiveSong.title,
        url = ActiveSong.url,
        isPlaying = true,
        volume = ActiveSong.volume,
    })

    cb('ok')
end)

RegisterNUICallback("TOGGLE_PLAYING", function(_, cb)
    if not ActiveSong or not ActiveSong.url then
        cb('ok')
        return
    end

    if GetResourceState('xsound') ~= 'started' then
        cb('ok')
        return
    end

    local xsound = exports.xsound
    local songId = "sm_chatheme-song-" .. cache.serverId

    if not xsound:soundExists(songId) then
        cb('ok')
        return
    end

    local isPaused = xsound:isPaused(songId)

    if isPaused then
        xsound:Resume(songId)
    else
        xsound:Pause(songId)
    end

    local newData = xsound:getInfo(songId)
    SendVueMessage("SONG_DATA", {
        title = ActiveSong.title,
        url = ActiveSong.url,
        isPlaying = isPaused,
        volume = newData.volume,
    })

    cb('ok')
end)

RegisterNUICallback("STOP_MUSIC", function(_, cb)
    if ActiveSong and ActiveSong.url and GetResourceState('xsound') == 'started' then
        local xsound = exports.xsound
        local songId = "sm_chatheme-song-" .. cache.serverId

        if xsound:soundExists(songId) then
            xsound:Destroy(songId)
            ActiveSong.isPlaying = false
        end
    end

    cb('ok')
end)

RegisterNUICallback("UPDATE_VOLUME", function(data, cb)
    if ActiveSong and ActiveSong.url and GetResourceState('xsound') == 'started' then
        local xsound = exports.xsound
        local songId = "sm_chatheme-song-" .. cache.serverId

        if xsound:soundExists(songId) then
            local volume = data.volume and tonumber(data.volume) / 100 or Config.MusicSettings.defaultVolume
            xsound:setVolume(songId, volume)
        end
    end

    cb('ok')
end)

local Config = require "config.config"

local CreateThread = CreateThread
local Wait = Wait
local GetEntityCoords = GetEntityCoords
local GetPlayerPed = GetPlayerPed
local GetPlayerFromServerId = GetPlayerFromServerId
local GetHudScreenPositionFromWorldPosition = GetHudScreenPositionFromWorldPosition
local mathAbs = math.abs
local tonumber = tonumber
local next = next
local pairs = pairs

local MaxDistance = 10.0
local MoveThreshold = 0.05
local DefaultColor = { r = 255, g = 255, b = 255, a = 255 }
local DrawColors = Config.DefaultSettings.drawtextColors
local RemovePayload = {}

---@param text table
local function HideDUIText(text)
    if text.hidden then return end
    text.hidden = true
    text.oldScreenX = nil
    text.oldScreenY = nil
    RemovePayload.source = text.source
    SendVueMessage("REMOVE_3D_TEXT", RemovePayload)
end

function RPTextsDUIThread()
    if RPTextThread then return end
    RPTextThread = true

    CreateThread(function()
        while #RPTexts >= 1 do
            local playerPedCoords = GetEntityCoords(cache.ped)

            for i = 1, #RPTexts do
                local text = RPTexts[i]
                local playerId = text.source and GetPlayerFromServerId(text.source) or -1

                if playerId == -1 then
                    HideDUIText(text)
                else
                    local targetPedCoords = GetEntityCoords(GetPlayerPed(playerId))

                    if #(playerPedCoords - targetPedCoords) >= MaxDistance then
                        HideDUIText(text)
                    else
                        local offscreen, screenX, screenY = GetHudScreenPositionFromWorldPosition(
                            targetPedCoords.x, targetPedCoords.y, targetPedCoords.z + 1.0
                        )

                        if offscreen then
                            HideDUIText(text)
                        else
                            screenX = screenX * 100
                            screenY = screenY * 100
                            local oldX, oldY = text.oldScreenX, text.oldScreenY

                            if not oldX or mathAbs(oldX - screenX) > MoveThreshold or mathAbs(oldY - screenY) > MoveThreshold then
                                text.hidden = false
                                text.oldScreenX = screenX
                                text.oldScreenY = screenY
                                SendVueMessage('SHOW_3D_TEXT', {
                                    source = text.source,
                                    player = text.fullname,
                                    screenX = screenX,
                                    screenY = screenY,
                                    message = text.message,
                                    category = text.category,
                                    isDice = text.isDice,
                                    isRPS = text.isRPS,
                                    hand = text.hand or false,
                                })
                            end
                        end
                    end
                end
            end
            Wait(0)
        end
        RPTextThread = false
    end)
end

function RPTexts3DThread()
    if RPText3DThread then return end
    RPText3DThread = true

    CreateThread(function()
        while #RPTexts >= 1 do
            local playerPedCoords = GetEntityCoords(cache.ped)

            for i = 1, #RPTexts do
                local text = RPTexts[i]
                local playerId = text.source and GetPlayerFromServerId(text.source) or -1

                if playerId ~= -1 then
                    local targetPedCoords = GetEntityCoords(GetPlayerPed(playerId))

                    if #(playerPedCoords - targetPedCoords) < MaxDistance then
                        DrawText3D(targetPedCoords.x, targetPedCoords.y, targetPedCoords.z, text.message, DrawColors[text.category] or DefaultColor)
                    end
                end
            end
            Wait(0)
        end
        RPText3DThread = false
    end)
end

function FWritingEffect()
    if WritingEffect then return end
    WritingEffect = true

    CreateThread(function()
        while next(WritingPlayers) ~= nil do
            local sleep = 500
            local playerCoords = GetEntityCoords(cache.ped)

            for source in pairs(WritingPlayers) do
                local writerId = GetPlayerFromServerId(tonumber(source))
                if writerId == -1 then
                    WritingPlayers[source] = nil
                else
                    local writerCoords = GetEntityCoords(GetPlayerPed(writerId))
                    if #(playerCoords - writerCoords) <= 6.0 then
                        sleep = 0
                        DrawDots(vec3(writerCoords.x, writerCoords.y, writerCoords.z + 1.0), Dots)
                    end
                end
            end
            Wait(sleep)
        end
        WritingEffect = false
    end)
end

local Config = require 'config.config'

local function LoadLocaleFile()
    local path = 'locales/' .. Config.Locale .. '.json'
    local file = LoadResourceFile(GetCurrentResourceName(), path)

    if not file then
        file = LoadResourceFile(GetCurrentResourceName(), 'locales/en.json')
        print(('Locale file "%s" not found, falling back to default "en".'):format(Config.Locale))
    end

    return file
end

local function LoadLocale()
    local file = LoadLocaleFile()

	if not file or file == '' then
        error('Locale file is empty or not found')
    end

    local success, content = pcall(json.decode, file)

	if not success then
        error(('Failed to decode JSON from locale file: %s'):format(content))
    end


    if not content['Lua'] then
        error('Missing "Lua" section in locale file')
    end

    return content['Lua']
end

Locales = LoadLocale()

function _T(key, ...)
    if Locales[key] then
        return Locales[key]:format(...)
    end
    return key
end
