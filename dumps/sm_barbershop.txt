########## FILE: \modules\nui.lua
local open = {}

local Nui = {}

local function refresh()
    local focused = next(open) ~= nil

    SetNuiFocus(focused, focused)
end

---@param name string
---@param action string
---@param data any
function Nui.open(name, action, data)
    open[name] = true
    SendNUIMessage({ action = action, data = data })
    refresh()
end

---@param name string
---@param action string
function Nui.close(name, action)
    if not open[name] then return end

    open[name] = nil
    SendNUIMessage({ action = action })
    refresh()
end

---@param name string
---@return boolean
function Nui.isOpen(name)
    return open[name] == true
end

---@param name string
---@param handler fun(payload: table): any
function Nui.callback(name, handler)
    RegisterNUICallback(name, function(payload, cb)
        local ok, result = pcall(handler, payload or {})

        if not ok then
            lib.print.error(('nui callback %s failed: %s'):format(name, result))
            cb({})

            return
        end

        cb(result == nil and {} or result)
    end)
end

AddEventHandler('onResourceStop', function(resource)
    if resource ~= cache.resource then return end

    SetNuiFocus(false, false)
end)

return Nui

local config = Config

local Pricing = {}

local OPACITY_EPSILON = 0.001

local MIN_VALUES = {
    style = -1,
    texture = 0,
    color = 0,
    highlight = 0,
    secondColor = 0,
    fade = -1,
    eyeColor = 0,
}

---@type table<string, BarberPriceItem>
local byKey = {}

for i = 1, #config.priceItems do
    local item = config.priceItems[i]

    byKey[item.key] = item
end

Pricing.byKey = byKey

---@type table<string, BarberPriceGroup>
local groups = {}

for i = 1, #config.priceGroups do
    groups[config.priceGroups[i].id] = config.priceGroups[i]
end

---@param key string
---@return integer min
---@return integer max
function Pricing.bounds(key)
    local item = byKey[key]
    local group = item and groups[item.group]

    if not group then return 0, 0 end

    return group.min, group.max
end

---@param key string
---@return integer
function Pricing.defaultPrice(key)
    local min, max = Pricing.bounds(key)

    return math.floor((min + max) / 2)
end

---@param key string
---@return string section
---@return string? field
local function split(key)
    local section, field = key:match('^(%w+)%.(%w+)$')

    if section then
        return section, field
    end

    return key
end

---@param appearance table
---@param key string
---@return number?
function Pricing.read(appearance, key)
    local section, field = split(key)

    if not field then
        return tonumber(appearance[section])
    end

    local source = section == 'hair' and appearance.hair or (appearance.headOverlays and appearance.headOverlays[section])

    return source and tonumber(source[field]) or nil
end

---@param appearance table
---@param key string
---@param value number
function Pricing.write(appearance, key, value)
    local section, field = split(key)

    if not field then
        appearance[section] = value

        return
    end

    if section == 'hair' then
        appearance.hair = appearance.hair or {}
        appearance.hair[field] = value

        return
    end

    appearance.headOverlays = appearance.headOverlays or {}
    appearance.headOverlays[section] = appearance.headOverlays[section] or { style = 0, opacity = 0.0, color = 0, secondColor = 0 }
    appearance.headOverlays[section][field] = value
end

---@param field string
---@param value any
---@return number?
local function normalizeField(field, value)
    local number = tonumber(value)

    if not number or number ~= number then return end

    if field == 'opacity' then
        return math.floor(math.min(1, math.max(0, number)) * 100 + 0.5) / 100
    end

    local min = MIN_VALUES[field]
    local integer = min and math.tointeger(math.floor(number))

    if not integer or integer < min then return end

    return integer
end

---@param key string
---@param value any
---@return number?
function Pricing.normalize(key, value)
    if not byKey[key] then return end

    local _, field = split(key)

    return normalizeField(field or key, value)
end

local HAIR_FIELDS = { 'style', 'texture', 'color', 'highlight', 'fade' }
local OVERLAY_FIELDS = { 'style', 'opacity', 'color', 'secondColor' }

local OVERLAYS = {
    blemishes = true,
    beard = true,
    eyebrows = true,
    ageing = true,
    makeUp = true,
    blush = true,
    complexion = true,
    sunDamage = true,
    lipstick = true,
    moleAndFreckles = true,
    chestHair = true,
    bodyBlemishes = true,
}

---@param values any
---@return BarberAppearanceFields?
function Pricing.sanitizePreview(values)
    if type(values) ~= 'table' then return end

    local clean, any = {}, false

    if values.hair ~= nil then
        if type(values.hair) ~= 'table' then return end

        local hair = {}

        for i = 1, #HAIR_FIELDS do
            local field = HAIR_FIELDS[i]
            local value = normalizeField(field, values.hair[field])

            if value == nil then return end

            hair[field] = value
        end

        clean.hair, any = hair, true
    end

    if values.headOverlays ~= nil then
        if type(values.headOverlays) ~= 'table' then return end

        local overlays = {}

        for name, overlay in pairs(values.headOverlays) do
            if not OVERLAYS[name] or type(overlay) ~= 'table' then return end

            local entry = {}

            for i = 1, #OVERLAY_FIELDS do
                local field = OVERLAY_FIELDS[i]
                local value = normalizeField(field, overlay[field])

                if value == nil then return end

                entry[field] = value
            end

            overlays[name] = entry
            any = true
        end

        clean.headOverlays = overlays
    end

    if values.eyeColor ~= nil then
        local value = normalizeField('eyeColor', values.eyeColor)

        if value == nil then return end

        clean.eyeColor, any = value, true
    end

    return any and clean or nil
end

---@param appearance table
---@return BarberAppearanceFields
function Pricing.barberFields(appearance)
    return {
        hair = appearance.hair,
        headOverlays = appearance.headOverlays,
        eyeColor = appearance.eyeColor,
    }
end

---@param items BarberLineItem[]
---@return { label: string, price: integer }[]
function Pricing.describe(items)
    local lines = {}

    for i = 1, #items do
        local item = items[i]
        local group = groups[item.group] and groups[item.group].label or item.group

        lines[i] = {
            label = item.key == item.group and group or ('%s Â· %s'):format(group, item.label),
            price = item.price,
        }
    end

    return lines
end

---@param values any
---@return table<string, number>
function Pricing.sanitize(values)
    local clean = {}

    if type(values) ~= 'table' then return clean end

    for key, value in pairs(values) do
        if type(key) == 'string' then
            clean[key] = Pricing.normalize(key, value)
        end
    end

    return clean
end

---@param key string
---@param before number?
---@param after number?
---@return boolean
local function differs(key, before, after)
    if before == nil or after == nil then
        return before ~= after
    end

    if key:sub(-8) == '.opacity' then
        return math.abs(before - after) > OPACITY_EPSILON
    end

    return before ~= after
end

---@param original table
---@param values table<string, number>
---@param prices table<string, number>
---@return table final
---@return BarberLineItem[] items
---@return integer total
function Pricing.apply(original, values, prices)
    local final = lib.table.deepclone(original)
    local changed = {}

    for i = 1, #config.priceItems do
        local key = config.priceItems[i].key
        local value = values[key]

        if value ~= nil and differs(key, Pricing.read(original, key), value) then
            Pricing.write(final, key, value)
            changed[key] = true
        end
    end

    local items, total = {}, 0

    for i = 1, #config.priceItems do
        local item = config.priceItems[i]

        if changed[item.key] then
            local price = (item.includedWith and changed[item.includedWith]) and 0 or math.floor(prices[item.key] or 0)

            items[#items + 1] = { key = item.key, group = item.group, label = item.label, price = price }
            total += price
        end
    end

    return final, items, total
end

local CUT_GROUPS = { hair = true, beard = true, eyebrows = true, chestHair = true }

---@param items BarberLineItem[]
---@return boolean

---@param search string?
---@return string?
local function likePattern(search)
    if type(search) ~= 'string' or search == '' then return end

    return '%' .. search:gsub('[\\%%_]', '\\%0') .. '%'
end

---@param items table[]
---@param page integer
---@param pageSize integer
---@param total integer
---@return table
local function pageOf(items, page, pageSize, total)
    return {
        items = items,
        page = page,
        pageSize = pageSize,
        total = total,
        numberOfPages = math.max(1, math.ceil(total / pageSize)),
    }
end

---@param tableName string
---@param column string
---@return boolean
local function columnExists(tableName, column)
    local count = tonumber(db:scalar_async([[
        SELECT COUNT(*) FROM information_schema.COLUMNS
        WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = ? AND COLUMN_NAME = ?
    ]], { tableName, column })) or 0

    return count > 0
end

---@param tableName string
---@param column string
---@param definition string
local function ensureColumn(tableName, column, definition)
    if not columnExists(tableName, column) then
        db:query_async(('ALTER TABLE `%s` ADD COLUMN `%s` %s'):format(tableName, column, definition))
    end
end

---@param tableName string
---@param column string
local function dropColumn(tableName, column)
    if columnExists(tableName, column) then
        db:query_async(('ALTER TABLE `%s` DROP COLUMN `%s`'):format(tableName, column))
    end
end

---@param shopId string
---@param page integer
---@param pageSize integer
---@param search string?
---@return BarberServicePage
function storage.getServiceLogs(shopId, page, pageSize, search)
    local pattern = likePattern(search)
    local filter = pattern and ' AND (customer_name LIKE ? OR employee_name LIKE ?)' or ''
    local params = pattern and { shopId, pattern, pattern } or { shopId }

    local total = tonumber(db:scalar_async(
        'SELECT COUNT(*) FROM `sm_barbershop_services` WHERE shop_id = ?' .. filter,
        params
    )) or 0

    local rows = db:query_async(([[
        SELECT id, mode, customer_name, employee_name, amount, commission,
            DATE_FORMAT(created_at, '%%Y-%%m-%%d') AS date, DATE_FORMAT(created_at, '%%H:%%i') AS at
        FROM `sm_barbershop_services`
        WHERE shop_id = ?%s
        ORDER BY id DESC
        LIMIT ? OFFSET ?
    ]]):format(filter), pattern
        and { shopId, pattern, pattern, pageSize, (page - 1) * pageSize }
        or { shopId, pageSize, (page - 1) * pageSize })

    local items = {}

    if type(rows) == 'table' then
        for i = 1, #rows do
            items[i] = mapService(rows[i])
        end
    end

    return pageOf(items, page, pageSize, total)
end

local Text = {}

---@param value any
---@param min integer
---@param max integer
---@return string?
function Text.name(value, min, max)
    if type(value) ~= 'string' then return end

    local name = value:gsub('%s+', ' '):match('^%s*(.-)%s*$')

    if #name < min or #name > max or not name:match("^[%w %.'&%-]+$") then return end

    return name
end

local syncBaseline

---@type table<string, integer>
local syncApplied = {}

---@type BarberManagementSync[]
local pendingSync = {}

---@param baseline integer?
local function resetSync(baseline)
    syncBaseline = baseline
    syncApplied = {}
    pendingSync = {}
end

---@param payload BarberManagementSync
---@param baseline integer
---@return BarberManagementSync?
local function freshFields(payload, baseline)
    local revision = payload.revision
    local fresh

    for key, value in pairs(payload) do
        if key ~= 'shopId' and key ~= 'revision' and revision > (syncApplied[key] or baseline) then
            syncApplied[key] = revision
            fresh = fresh or { shopId = payload.shopId }
            fresh[key] = value
        end
    end

    return fresh
end

Nui.callback('barber:management:bootstrap', function()
    local result = serverAwait('bootstrap')

    if type(result) ~= 'table' or type(result.revision) ~= 'number' then return result end

    local pending = pendingSync

    resetSync(result.revision)

    for i = 1, #pending do
        local fresh = freshFields(pending[i], result.revision)

        if fresh then
            fresh.shopId = nil

            for key, value in pairs(fresh) do
                result[key] = value
            end
        end
    end

    return result
end)

---@param shopId string
local function buildDaily(shopId)
    local byDate = Storage.getDailyStats(shopId, 7)
    local series = {}
    local now = os.time()

    for offset = 6, 0, -1 do
        local ts = now - offset * 86400
        local entry = byDate[os.date('%Y-%m-%d', ts)]

        series[#series + 1] = {
            label = DAY_LABELS[os.date('*t', ts).wday],
            visitors = entry and entry.visitors or 0,
            revenue = entry and entry.revenue or 0,
        }
    end

    return series
end

local function pageOf(value)
    return math.max(1, math.floor(tonumber(value) or 1))

local function pageSizeOf(value)
    local size = math.floor(tonumber(value) or limits.logPageSize)

    return math.min(limits.maxPageSize, math.max(1, size))
end

local function searchOf(value)
    if type(value) ~= 'string' then return end

    local term = value:match('^%s*(.-)%s*$'):sub(1, 40)

    return term ~= '' and term or nil
end

local function amountOf(value)
    local amount = tonumber(value)

    if not amount or amount ~= amount then return end

    amount = math.floor(amount)

    if amount <= 0 or amount > limits.maxFinanceAmount then return end

    return amount
end

lib.callback.register('sm_barbershop:management:buySupply', function(source, shopId, itemName, quantity)

    if count < 1 or count > supply.maxQuantity then return fail(('Jumlah harus 1-%d.'):format(supply.maxQuantity)) end

    if total > shop.balance then return fail('Saldo barbershop tidak cukup.') end

    if not player then return fail('Data pemain tidak ditemukan.') end
    local metadata = Wear.metadata(supply.item, Wear.max(supply.item))

    if not inventory:AddItem(source, supply.item, count, metadata) then return fail('Barang gagal dimasukkan ke inventory.') end

    Shops.addBalance(shop, -total)
    Storage.addTransaction({
        shopId = shop.id,
        type = 'supply',
        amount = total,
        balanceAfter = shop.balance,
        actorCitizenId = Staff.citizenIdOf(player),
        actorName = Staff.nameOf(player),
        note = ('Beli %dx %s'):format(count, supply.label),
    })
    syncShop(shop)

    return { success = true, shop = Shops.publicData(shop) }
end)

local LOCATE_RADIUS = 30.0
local COMMAND_USAGE = 'Pakai: /barbershop setowner [playerid] [barberid] | removeowner [barberid] | getid | goto [barberid]'

---@type table<string, vector3>
local managementOf = {}

for i = 1, #config.shops do
    managementOf[config.shops[i].id] = config.shops[i].coords.management
end

---@param source number
---@param notifyType string
---@param message string
local function reply(source, notifyType, message)
    if source == 0 then
        return print(message)
    end

    notify(source, notifyType, message)
end

---@param source number
---@return BarberShop?
local function shopNear(source)
    local ped = GetPlayerPed(source)

    if ped == 0 then return end

    local coords = GetEntityCoords(ped)
    local best, bestDistance

    for id, management in pairs(managementOf) do
        local distance = #(coords - management)

        if distance <= LOCATE_RADIUS and (not bestDistance or distance < bestDistance) then
            best, bestDistance = id, distance
        end
    end

    return best and Shops.get(best) or nil
end

---@type table<string, fun(source: number, first: string?, second: string?)>
local commands = {}

commands.setowner = function(source, playerId, shopId)
    local shop = Shops.get(shopId)

    if not shop then return reply(source, 'error', 'ID barbershop tidak ditemukan.') end

    local target = tonumber(playerId)
    local player = target and Staff.getPlayer(target)
    local citizenId = Staff.citizenIdOf(player)

    if not target or not citizenId then return reply(source, 'error', 'Pemain target tidak ditemukan.') end

    local previousOwner = shop.ownerCitizenId

    Shops.setOwner(shop, citizenId, Staff.nameOf(player))
    Staff.syncPlayer(target)

    if previousOwner and previousOwner ~= citizenId then
        Staff.syncCitizen(previousOwner)
    end

    syncShop(shop)
    reply(source, 'success', ('%s sekarang pemilik %s.'):format(Staff.nameOf(player), shop.label))
end

commands.removeowner = function(source, shopId)
    local shop = Shops.get(shopId)

    if not shop then return reply(source, 'error', 'ID barbershop tidak ditemukan.') end

    local previousOwner = shop.ownerCitizenId

    if not previousOwner then return reply(source, 'error', ('%s belum punya pemilik.'):format(shop.label)) end

    Shops.setOwner(shop, nil, nil)
    Staff.syncCitizen(previousOwner)
    syncShop(shop)
    reply(source, 'success', ('Pemilik %s dihapus.'):format(shop.label))
end

commands.getid = function(source)
    if source == 0 then return reply(source, 'error', 'Command ini hanya bisa dipakai di dalam game.') end

    local shop = shopNear(source)

    if not shop then return reply(source, 'error', 'Anda tidak berada di area barbershop.') end

    reply(source, 'inform', ('ID barbershop: %s (%s)'):format(shop.id, shop.label))
end

commands['goto'] = function(source, shopId)
    if source == 0 then return reply(source, 'error', 'Command ini hanya bisa dipakai di dalam game.') end

    local management = type(shopId) == 'string' and managementOf[shopId]

    if not management then return reply(source, 'error', 'ID barbershop tidak ditemukan.') end

    local ped = GetPlayerPed(source)

    if ped == 0 then return end

    SetEntityCoords(ped, management.x, management.y, management.z, false, false, false, false)
end

---@return vector3? coords
---@return number? heading
local function choosePlacement()
    lib.requestModel(model)

    local ped = cache.ped
    local start = GetEntityCoords(ped)
    local ghost = CreateObjectNoOffset(model, start.x, start.y, start.z, false, false, false)
    local heading = GetEntityHeading(ped)
    local chosen

    SetModelAsNoLongerNeeded(model)
    SetEntityAlpha(ghost, GHOST_ALPHA, false)
    SetEntityCollision(ghost, false, false)
    FreezeEntityPosition(ghost, true)
    SetEntityDrawOutline(ghost, true)
    lib.showTextUI(HINT, { position = 'right-center' })

    while not (cache.vehicle or IsEntityDead(ped) or not LocalPlayer.state.barberDuty) do
        for i = 1, #BLOCKED_CONTROLS do
            DisableControlAction(0, BLOCKED_CONTROLS[i], true)
        end

        local hit, _, target, normal = lib.raycast.fromCamera(RAY_FLAGS, 4, settings.reach + RAY_EXTRA)
        local valid = hit and normal.z >= MIN_NORMAL_Z and #(target - GetEntityCoords(ped)) <= settings.reach and not crowded(target)

        if hit then
            SetEntityCoordsNoOffset(ghost, target.x, target.y, target.z, false, false, false)
        end

        SetEntityHeading(ghost, heading)
        SetEntityDrawOutlineColor(valid and 0 or 255, valid and 255 or 0, 0, 255)

        if IsDisabledControlPressed(0, ROTATE_LEFT_CONTROL) then
            heading = (heading + ROTATE_SPEED * GetFrameTime()) % 360.0
        elseif IsDisabledControlPressed(0, ROTATE_RIGHT_CONTROL) then
            heading = (heading - ROTATE_SPEED * GetFrameTime()) % 360.0
        end

        if IsDisabledControlJustPressed(0, CANCEL_CONTROL) then break end

        if valid and IsDisabledControlJustPressed(0, PLACE_CONTROL) then
            chosen = target
            break
        end

        Wait(0)
    end

    lib.hideTextUI()
    SetEntityDrawOutline(ghost, false)
    DeleteEntity(ghost)

    return chosen, chosen and heading or nil
end

---@param list any
---@return table<string, boolean>?
local function permissionSet(list)
    if type(list) ~= 'table' then return end

    local set = {}

    for i = 1, #list do
        local key = list[i]

        if type(key) ~= 'string' or not known[key] then return end

        set[key] = true
    end

    return set
end

---@param set table<string, boolean>
---@return string[]
local function permissionList(set)
    local list = {}

    for i = 1, #config.permissions do
        local key = config.permissions[i].key

        if set[key] then
            list[#list + 1] = key
        end
    end

    return list
end

---@param raw any
---@return table<string, boolean>
local function decode(raw)
    local ok, decoded = pcall(json.decode, type(raw) == 'string' and raw or 'null')

    return ok and permissionSet(decoded) or {}
end

---@param shopId string
---@return BarberRole[]
local function listOf(shopId)
    local list = byShop[shopId]

    if not list then
        list = {}
        byShop[shopId] = list
    end

    return list
end

function Roles.init()
    local rows = Storage.loadRoles()

    for i = 1, #rows do
        local row = rows[i]
        local list = listOf(row.shop_id)

        list[#list + 1] = {
            id = row.id,
            key = row.role_key or ('r%d'):format(row.id),
            label = row.label,
            permissions = decode(row.permissions),
        }
    end

    for i = 1, #config.shops do
        local list = listOf(config.shops[i].id)

        if #list == 0 then
            for j = 1, #config.roles do
                local default = config.roles[j]
                local permissions = permissionSet(default.permissions) or {}
                local id = Storage.addRole(config.shops[i].id, default.key, default.label, json.encode(permissionList(permissions)))

                if id then
                    list[#list + 1] = { id = id, key = default.key, label = default.label, permissions = permissions }
                end
            end
        end
    end
end

---@return BarberRole[]

    return listOf(shopId)
end

---@param shopId string
---@param key string?
---@param label any
---@param permissions any
---@return BarberRole?
---@return string?
function Roles.save(shopId, key, label, permissions)
    local name = Text.name(label, limits.minRoleName, limits.maxRoleName)

    if not name then
        return nil, ('Nama role harus %d-%d karakter, hanya huruf, angka, spasi, dan . \' & -'):format(limits.minRoleName, limits.maxRoleName)
    end

    local set = permissionSet(permissions)

    if not set then return nil, 'Izin tidak valid.' end

    local list = listOf(shopId)

    for i = 1, #list do
        if list[i].key ~= key and list[i].label:lower() == name:lower() then
            return nil, 'Nama role sudah dipakai.'
        end
    end

    local encoded = json.encode(permissionList(set))

    if key then
        local role = Roles.get(shopId, key)

        if not role then return nil, 'Role tidak ditemukan.' end

        role.label = name
        role.permissions = set
        Storage.saveRole(role.id, name, encoded)

        return role
    end

    if #list >= limits.maxRoles then return nil, ('Maksimal %d role.'):format(limits.maxRoles) end

    local id = Storage.addRole(shopId, nil, name, encoded)

    if not id then return nil, 'Role gagal disimpan.' end

    ---@type BarberRole
    local role = { id = id, key = ('r%d'):format(id), label = name, permissions = set }

    list[#list + 1] = role

    return role
end

---@param shopId string
---@param key string
---@return boolean
function Roles.remove(shopId, key)
    local list = listOf(shopId)

    for i = 1, #list do
        if list[i].key == key then
            Storage.removeRole(list[i].id)
            table.remove(list, i)

            return true
        end
    end

    return false
end

return Roles

---@field keeper number
---@field networked boolean
---@field scissors number?
---@field customerId number?
---@field customerPed number?
---@field result (BarberServiceStart|BarberSelfStart)?
---@field stage BarberServiceStage
---@field entered boolean?
---@field attached boolean?
---@field ready boolean?
---@field arrived boolean?

---@type BarberChairEntry[]
local chairs = {}

---@type table<string, BarberAnimationSet>
local shopAnimation = {}

for i = 1, #config.shops do
    local shop = config.shops[i]

    shopAnimation[shop.id] = animation.sets[shop.animation or 'barbers']

    for j = 1, #shop.coords.chairs do
        chairs[#chairs + 1] = { shop = shop.id, index = j, coords = shop.coords.chairs[j] }
    end
end

---@type BarberSeat?
local customer

---@type BarberService?
local service

local busy = false
local exiting = false

local function notify(notifyType, description)
    lib.notify({ type = notifyType, description = description })
end

---@param list string[]
---@return string
local function pick(list)
    return list[math.random(#list)]
end

---@param entry BarberChairEntry
---@return BarberSceneOrigin
local function sceneOrigin(entry)
    local coords = entry.coords
    local set = entry.portable and animation.sets[config.portable.animation] or shopAnimation[entry.shop] or animation.sets.barbers

    return {
        pos = coords.xyz,
        heading = coords.w + config.scene.headingOffset,
        dict = set.dict,
        set = set,
    }
end

---@param origin BarberSceneOrigin
---@param clip string
---@return vector3 position
---@return number heading
local function clipStart(origin, clip)
    local pos = origin.pos
    local start = GetAnimInitialOffsetPosition(origin.dict, clip, pos.x, pos.y, pos.z, 0.0, 0.0, origin.heading, 0.0, ROTATION_ORDER)
    local rotation = GetAnimInitialOffsetRotation(origin.dict, clip, pos.x, pos.y, pos.z, 0.0, 0.0, origin.heading, 0.0, ROTATION_ORDER)

    return start, rotation.z
end

---@param origin BarberSceneOrigin
---@param clip string
---@return integer
local function clipDuration(origin, clip)
    return math.floor(GetAnimDuration(origin.dict, clip) * 1000)
end

---@param clip string
---@return string
local function propClip(clip)
    return (clip:gsub('^keeper_', 'scissors_'))
end

---@param list BarberChairEntry[]
---@param coords vector3
---@param best BarberChairEntry?
---@param bestDistance number?
---@return BarberChairEntry?
---@return number?
local function closer(list, coords, best, bestDistance)
    for i = 1, #list do
        local entry = list[i]
        local distance = #(coords - entry.coords.xyz)

        if distance <= config.chairDistance and (not bestDistance or distance < bestDistance) then
            best, bestDistance = entry, distance
        end
    end

    return best, bestDistance
end

---@param coords vector3
---@return BarberChairEntry?
local function nearestChair(coords)
    local best, bestDistance = closer(chairs, coords)

    return (closer(Sync.portables(), coords, best, bestDistance))
end

---@param entry BarberChairEntry
---@return vector3?
local function seatPosition(entry)
    if entry.seat then return entry.seat end

    local origin = sceneOrigin(entry)

    if not HasAnimDictLoaded(origin.dict) then
        RequestAnimDict(origin.dict)

        return
    end

    entry.seat = clipStart(origin, origin.set.seated[1])

    return entry.seat
end

---@param entry BarberChairEntry
---@return boolean
local function chairFree(entry)
    if Sync.chair(entry.shop, entry.index) then return false end

    local seat = seatPosition(entry)

    if not seat then return true end

    if #lib.getNearbyPlayers(seat, SEAT_RADIUS, false) > 0 then
        return false
    end

    return lib.getClosestPed(seat, SEAT_RADIUS) == nil
end

---@param ped number
---@param origin BarberSceneOrigin
---@param clip string
---@param stage BarberSceneStage
---@param prop number?
---@return integer netScene
local function playNetworkScene(ped, origin, clip, stage, prop)
    local pos = origin.pos
    local scene = NetworkCreateSynchronisedScene(pos.x, pos.y, pos.z, 0.0, 0.0, origin.heading, ROTATION_ORDER, stage.hold, stage.loop, 1.0, 0.0, 1.0)

    NetworkAddPedToSynchronisedScene(ped, scene, origin.dict, clip, stage.blendIn, stage.blendOut, stage.flags, 0, stage.mover or INSTANT_BLEND, 0)

    if prop and DoesEntityExist(prop) then
        FreezeEntityPosition(prop, false)
        NetworkAddEntityToSynchronisedScene(prop, scene, origin.dict, propClip(clip), INSTANT_BLEND, -INSTANT_BLEND, stage.propFlags or 0)
    end

    NetworkStartSynchronisedScene(scene)

    return scene
end

---@param scene integer
---@param networked boolean
---@param phase number
---@param timeout integer
---@param active (fun(): boolean)?
---@return boolean reached
local function waitScenePhase(scene, networked, phase, timeout, active)
    local deadline = GetGameTimer() + timeout
    local started = false

    while GetGameTimer() < deadline and (not active or active()) do
        local id = networked and NetworkGetLocalSceneFromNetworkId(scene) or scene

        if id ~= -1 and IsSynchronizedSceneRunning(id) then
            started = true

            if GetSynchronizedScenePhase(id) >= phase then return true end
        elseif started then
            return false
        end

        Wait(0)
    end

    return false
end

---@param ped number
---@param origin BarberSceneOrigin
---@param clip string
---@param stage BarberSceneStage
---@param prop number?
---@return integer scene
local function playLocalScene(ped, origin, clip, stage, prop)
    local pos = origin.pos
    local scene = CreateSynchronizedScene(pos.x, pos.y, pos.z, 0.0, 0.0, origin.heading, ROTATION_ORDER)

    SetSynchronizedSceneHoldLastFrame(scene, stage.hold)
    SetSynchronizedSceneLooped(scene, stage.loop)
    TaskSynchronizedScene(ped, scene, origin.dict, clip, stage.blendIn, stage.blendOut, stage.flags, 0, stage.mover or INSTANT_BLEND, 0)

    if prop and DoesEntityExist(prop) then
        FreezeEntityPosition(prop, false)
        PlaySynchronizedEntityAnim(prop, scene, propClip(clip), origin.dict, INSTANT_BLEND, -INSTANT_BLEND, 0, INSTANT_BLEND)
        ForceEntityAiAndAnimationUpdate(prop)
    end

    return scene
end

---@param current BarberService
---@param clip string
---@param stage BarberSceneStage
---@return integer? scene
local function playKeeper(current, clip, stage)
    if not DoesEntityExist(current.keeper) then return end

    local prop = not current.attached and current.scissors or nil

    if current.networked then
        return playNetworkScene(current.keeper, current.origin, clip, stage, prop)
    end

    return playLocalScene(current.keeper, current.origin, clip, stage, prop)
end

---@param current BarberService
---@param entry string?
---@return boolean ready
local function keeperEnter(current, entry)
    local origin = current.origin
    local networked = current.networked

    local function active()
        return service == current
    end

    current.entered = true

    if not entry then
        local base = playKeeper(current, 'keeper_base', KEEPER_STAGE.join)

        return base ~= nil and waitScenePhase(base, networked, 0.0, SCENE_START_GRACE, active)
    end

    local enter = playKeeper(current, entry, KEEPER_STAGE.enter)
    local timeout = clipDuration(origin, entry) + SCENE_START_GRACE

    if not enter or not waitScenePhase(enter, networked, SCENE_END_PHASE, timeout, active) then
        return false
    end

    local base = playKeeper(current, 'keeper_base', KEEPER_STAGE.base)

    return base ~= nil and waitScenePhase(base, networked, 0.0, SCENE_START_GRACE, active)
end

---@param ped number
---@param origin BarberSceneOrigin
---@param clip string
---@param timeout integer?
local function approach(ped, origin, clip, timeout)
    local target, heading = clipStart(origin, clip)

    timeout = timeout or APPROACH_TIMEOUT

    if #(target.xy - origin.pos.xy) > 3.0 or #(GetEntityCoords(ped).xy - target.xy) < 0.2 then return end

    TaskFollowNavMeshToCoord(ped, target.x, target.y, target.z, 1.0, timeout, 0.05, 0, heading)

    local deadline = GetGameTimer() + timeout

    repeat
        Wait(0)
    until GetScriptTaskStatus(ped, APPROACH_TASK) == TASK_FINISHED or GetGameTimer() > deadline
end

---@param entities number[]
---@param from integer
---@param to integer
local function fade(entities, from, to)
    for step = 1, FADE_STEPS do
        local alpha = math.floor(from + (to - from) * step / FADE_STEPS)

        for i = 1, #entities do
            if DoesEntityExist(entities[i]) then
                SetEntityAlpha(entities[i], alpha, false)
            end
        end

        Wait(FADE_MS // FADE_STEPS)
    end
end

local function lockPed(ped)
    SetPedCanRagdoll(ped, false)
    SetBlockingOfNonTemporaryEvents(ped, true)
end

local function releasePed(ped)
    FreezeEntityPosition(ped, false)
    SetPedCanRagdoll(ped, true)
    SetBlockingOfNonTemporaryEvents(ped, false)
end

local function unlockPed(ped)
    ClearPedTasks(ped)
    releasePed(ped)
end

---@param seat BarberSeat
local function revertCustomer(seat)
    local snapshot = seat.snapshot

    if snapshot then
        appearance:applyBarberValues({
            hair = snapshot.hair,
            headOverlays = snapshot.headOverlays,
            eyeColor = snapshot.eyeColor,
        })
    end

    appearance:setBarberHat(false)
end

---@param seat BarberSeat
local function sitScene(seat)
    local ped = cache.ped
    local origin = seat.origin

    playNetworkScene(ped, origin, 'player_enterchair', CUSTOMER_STAGE.enter)
    Wait(clipDuration(origin, 'player_enterchair'))

    if customer == seat then
        playNetworkScene(ped, origin, seat.seated, CUSTOMER_STAGE.seated)
    end
end

---@param origin BarberSceneOrigin
local function standScene(origin)
    local ped = cache.ped

    exiting = true

    local scene = playNetworkScene(ped, origin, 'player_exitchair', CUSTOMER_STAGE.exit)

    waitScenePhase(scene, true, config.scene.exitPhase, clipDuration(origin, 'player_exitchair'))
    NetworkStopSynchronisedScene(scene)
    releasePed(ped)
    exiting = false
end

local END_MESSAGES = {
    timeout = { 'warning', 'Sesi self service berakhir.' },
    chair_removed = { 'warning', 'Kursi cukur dibereskan, sesi berakhir.' },
}

local function announce(reason)
    local message = END_MESSAGES[reason]

    if message then
        notify(message[1], message[2])
    end
end

local function customerCleanup(reason)
    local seat = customer

    if not seat then return end

    customer = nil

    Nui.close('invoice', 'barber:invoice:close')
    lib.hideTextUI()
    appearance:closeBarberMenu()

    revertCustomer(seat)
    announce(reason)
    standScene(seat.origin)
end

local function leaveChair()
    if not customer then return end

    if lib.callback.await('sm_barbershop:chair:leave', false) then
        customerCleanup('left')
    end
end

local function seatLoop()
    local seat = customer

    while seat and customer == seat do
        for i = 1, #MOVEMENT_CONTROLS do
            DisableControlAction(0, MOVEMENT_CONTROLS[i], true)
        end

        if (seat.stage == 'waiting' or seat.stage == 'service') and IsControlJustReleased(0, INTERACT_KEY) then
            leaveChair()
        end

        Wait(0)
    end
end

---@param entry BarberChairEntry
local function sitDown(entry)
    if customer or service or busy or exiting then return end

    busy = true

    local result = lib.callback.await('sm_barbershop:chair:sit', false, entry.shop, entry.index)

    if not result.success then
        busy = false

        return notify('error', result.message)
    end

    local origin = sceneOrigin(entry)

    lib.requestAnimDict(origin.dict)

    ---@type BarberSeat
    local seat = { shop = entry.shop, chair = entry.index, origin = origin, seated = pick(origin.set.seated), stage = 'entering' }

    customer = seat
    lockPed(cache.ped)
    approach(cache.ped, origin, 'player_enterchair')

    if customer ~= seat then
        busy = false

        return
    end

    CreateThread(seatLoop)
    sitScene(seat)

    if customer ~= seat then
        busy = false

        return
    end

    seat.stage = 'waiting'

    local seated = lib.callback.await('sm_barbershop:chair:seated', false)

    busy = false

    if customer ~= seat then return end

    if not seated then return leaveChair() end

    lib.showTextUI(WAIT_TEXT, { position = 'right-center' })
end

---@param networked boolean
---@param coords vector3
---@return number
local function createScissors(networked, coords)
    local model = lib.requestModel(animation.scissors)
    local object = CreateObject(model, coords.x, coords.y, coords.z, networked, networked, false)

    SetModelAsNoLongerNeeded(model)
    SetEntityCollision(object, false, false)
    FreezeEntityPosition(object, true)

    if networked then
        local deadline = GetGameTimer() + 1000

        while not NetworkGetEntityIsNetworked(object) and GetGameTimer() < deadline do
            Wait(0)
        end
    end

    return object
end

---@class BarberGrip
---@field bone integer
---@field pos vector3
---@field rot vector3
---@field order integer

local HAND_BONES = { 57005, 18905 }
local GRIP_REACH = 0.35
local GRIP_TOLERANCE = 0.02
local ATTACH_FLAGS = { true, true, false, true }

---@type table<string, BarberGrip|false>
local grips = {}

---@param a vector3
---@param b vector3
---@return number
local function dot(a, b)
    return a.x * b.x + a.y * b.y + a.z * b.z
end

---@param v vector3
---@param bx vector3
---@param by vector3
---@param bz vector3
---@return vector3
local function intoBone(v, bx, by, bz)
    return vec3(dot(v, bx), dot(v, by), dot(v, bz))
end

---@param value number
---@return number
local function unit(value)
    return math.max(-1.0, math.min(1.0, value))
end

---@param entity number
---@return vector3 right
---@return vector3 forward
---@return vector3 up
---@return vector3 at
local function axesOf(entity)
    local at = GetEntityCoords(entity)

    return GetOffsetFromEntityInWorldCoords(entity, 1.0, 0.0, 0.0) - at,
        GetOffsetFromEntityInWorldCoords(entity, 0.0, 1.0, 0.0) - at,
        GetOffsetFromEntityInWorldCoords(entity, 0.0, 0.0, 1.0) - at,
        at
end

local EULER = {
    {
        order = 2,
        solve = function(r, f, u)
            return vec3(math.deg(math.asin(unit(f.z))), math.deg(math.atan(-r.z, u.z)), math.deg(math.atan(-f.x, f.y)))
        end,
    },
    {
        order = 0,
        solve = function(r, f, u)
            return vec3(math.deg(math.atan(f.z, u.z)), math.deg(math.asin(unit(-r.z))), math.deg(math.atan(r.y, r.x)))
        end,
    },
}

---@param entity number
---@param ped number
---@param boneIndex integer
---@param pos vector3
---@param rot vector3
---@param order integer
local function attachTo(entity, ped, boneIndex, pos, rot, order)
    AttachEntityToEntity(entity, ped, boneIndex, pos.x, pos.y, pos.z, rot.x, rot.y, rot.z,
        ATTACH_FLAGS[1], ATTACH_FLAGS[2], ATTACH_FLAGS[3], ATTACH_FLAGS[4], order, true)
end

---@param origin BarberSceneOrigin
---@return BarberGrip|false
local function measureGrip(origin)
    local pos = origin.pos
    local keeperModel = lib.requestModel(origin.set.keeper)
    local propModel = lib.requestModel(animation.scissors)
    local probe = CreatePed(4, keeperModel, pos.x, pos.y, pos.z, origin.heading, false, false)
    local prop = CreateObject(propModel, pos.x, pos.y, pos.z, false, false, false)
    local test = CreateObject(propModel, pos.x, pos.y, pos.z, false, false, false)

    SetModelAsNoLongerNeeded(keeperModel)
    SetModelAsNoLongerNeeded(propModel)

    for _, entity in ipairs({ probe, prop, test }) do
        SetEntityAlpha(entity, 0, false)
        SetEntityCollision(entity, false, false)
    end

    SetEntityInvincible(probe, true)
    SetBlockingOfNonTemporaryEvents(probe, true)

    local scene = CreateSynchronizedScene(pos.x, pos.y, pos.z, 0.0, 0.0, origin.heading, ROTATION_ORDER)

    SetSynchronizedSceneLooped(scene, true)
    TaskSynchronizedScene(probe, scene, origin.dict, 'keeper_base', INSTANT_BLEND, -INSTANT_BLEND, 0, 0, INSTANT_BLEND, 0)
    PlaySynchronizedEntityAnim(prop, scene, propClip('keeper_base'), origin.dict, INSTANT_BLEND, -INSTANT_BLEND, 0, INSTANT_BLEND)

    for _ = 1, 6 do
        ForcePedAiAndAnimationUpdate(probe, false, false)
        ForceEntityAiAndAnimationUpdate(prop)
        Wait(0)
    end

    SetSynchronizedSceneRate(scene, 0.0)
    Wait(0)

    local right, forward, up, at = axesOf(prop)
    local bone, reach = nil, GRIP_REACH

    for i = 1, #HAND_BONES do
        local distance = #(GetPedBoneCoords(probe, HAND_BONES[i], 0.0, 0.0, 0.0) - at)

        if distance < reach then
            bone, reach = HAND_BONES[i], distance
        end
    end

    ---@type BarberGrip|false
    local grip = false

    if bone then
        local base = GetPedBoneCoords(probe, bone, 0.0, 0.0, 0.0)
        local bx = GetPedBoneCoords(probe, bone, 1.0, 0.0, 0.0) - base
        local by = GetPedBoneCoords(probe, bone, 0.0, 1.0, 0.0) - base
        local bz = GetPedBoneCoords(probe, bone, 0.0, 0.0, 1.0) - base
        local offset = intoBone(at - base, bx, by, bz)
        local r, f, u = intoBone(right, bx, by, bz), intoBone(forward, bx, by, bz), intoBone(up, bx, by, bz)
        local boneIndex = GetPedBoneIndex(probe, bone)
        local bestError

        for i = 1, #EULER do
            local candidate = EULER[i]
            local rot = candidate.solve(r, f, u)

            attachTo(test, probe, boneIndex, offset, rot, candidate.order)
            Wait(0)

            local _, testForward, testUp, testAt = axesOf(test)
            local deviation = (1.0 - dot(testForward, forward)) + (1.0 - dot(testUp, up)) + #(testAt - at)

            if not bestError or deviation < bestError then
                bestError = deviation
                grip = { bone = bone, pos = offset, rot = rot, order = candidate.order }
            end

            DetachEntity(test, false, false)
        end

        if bestError and bestError > GRIP_TOLERANCE then
            lib.print.warn(('scissors grip for %s off by %.3f, falling back to the scene'):format(origin.dict, bestError))
            grip = false
        end
    end

    DeleteEntity(test)
    DeleteEntity(prop)
    DeleteEntity(probe)

    return grip
end

---@param origin BarberSceneOrigin
---@return BarberGrip|false
local function gripFor(origin)
    local grip = grips[origin.dict]

    if grip == nil then
        grip = measureGrip(origin)
        grips[origin.dict] = grip
    end

    return grip
end

---@type table<string, vector4>
local pedSpawns = {}

for i = 1, #config.shops do
    pedSpawns[config.shops[i].id] = config.shops[i].coords.pedspawn
end

---@param shopId string
---@param origin BarberSceneOrigin
---@return number npc
---@return number scissors
local function spawnKeeper(shopId, origin)
    local model = lib.requestModel(origin.set.keeper)
    local start, heading = clipStart(origin, 'keeper_enterchair')
    local spawn = pedSpawns[shopId]
    local coords = spawn and spawn.xyz or start
    local npc = CreatePed(4, model, coords.x, coords.y, coords.z, spawn and spawn.w or heading, false, false)

    SetModelAsNoLongerNeeded(model)
    SetEntityAlpha(npc, 0, false)
    SetEntityInvincible(npc, true)
    lockPed(npc)

    local scissors = createScissors(false, start)

    SetEntityAlpha(scissors, 0, false)

    return npc, scissors
end

---@param current BarberService
local function keeperWalk(current)
    local npc = current.keeper

    fade({ npc }, 0, 255)

    if DoesEntityExist(npc) then ResetEntityAlpha(npc) end

    approach(npc, current.origin, 'keeper_enterchair', KEEPER_WALK_TIMEOUT)

    if service ~= current then return end

    if current.scissors and DoesEntityExist(current.scissors) then
        ResetEntityAlpha(current.scissors)
    end

    current.ready = true
end

local function deleteEntity(entity)
    if entity and DoesEntityExist(entity) then
        DeleteEntity(entity)
    end
end

local function endBarberService()
    local current = service

    if not current or current.mode ~= 'barber' then return end

    service = nil
    exiting = true

    appearance:closeBarberMenu()

    local exit = current.entered and playKeeper(current, 'keeper_exitchair', KEEPER_STAGE.exit)

    if exit then
        waitScenePhase(exit, true, config.scene.exitPhase, clipDuration(current.origin, 'keeper_exitchair'))
        NetworkStopSynchronisedScene(exit)
    end

    deleteEntity(current.scissors)
    unlockPed(cache.ped)

    exiting = false
end

local function endSelfService(reason)
    local current = service
    local seat = customer

    if not current or current.mode ~= 'self' then return end

    service = nil
    customer = nil

    appearance:closeBarberMenu()
    announce(reason)

    local exit = current.entered and playKeeper(current, 'keeper_exitchair', KEEPER_STAGE.exit)

    if seat then
        standScene(seat.origin)
    end

    if exit then
        waitScenePhase(exit, false, config.scene.exitPhase, clipDuration(current.origin, 'keeper_exitchair'))
        SetSynchronizedSceneRate(exit, 0.0)
    end

    fade({ current.keeper, current.scissors --[[@as number]] }, 255, 0)
    deleteEntity(current.scissors)
    deleteEntity(current.keeper)
end

---@param current BarberService
---@param result table
---@param customerPed number
local function openServiceMenu(current, result, customerPed)
    local snapshot = result.snapshot

    local opened = appearance:openBarberMenu({
        target = customerPed,
        relay = true,
        appearance = snapshot.appearance,
        hairBlacklist = snapshot.hairBlacklist,
        hatWorn = snapshot.hatWorn,
        title = result.shopLabel,
        subtitle = result.customerName,
        prices = result.prices,
        bundles = BUNDLES,
        action = 'bill',
        onChange = function(values)
            TriggerServerEvent('sm_barbershop:service:preview', values)
        end,
        onHat = function(hidden)
            TriggerServerEvent('sm_barbershop:service:hat', hidden)
        end,
        onSubmit = function(values)
            CreateThread(function()
                local response = lib.callback.await('sm_barbershop:service:finish', false, values)

                if type(response) == 'table' and response.success and service == current then
                    current.stage = 'invoice'
                end

                appearance:resolveBarberMenu(response)
            end)
        end,
        onCancel = function()
            CreateThread(function()
                if service ~= current then return end

                lib.callback.await('sm_barbershop:service:cancel', false)
            end)
        end,
    })

    if not opened then
        lib.callback.await('sm_barbershop:service:cancel', false)

        return
    end

    TriggerServerEvent('sm_barbershop:service:ready')
end

---@param entry BarberChairEntry
local function startService(entry)
    if customer or service or busy or exiting then return end

    busy = true

    local result = lib.callback.await('sm_barbershop:service:start', false, entry.shop, entry.index)

    if not result.success then
        busy = false

        return notify('error', result.message)
    end

    local player = GetPlayerFromServerId(result.customerId)
    local ped = player ~= -1 and GetPlayerPed(player) or 0

    if ped == 0 then
        busy = false
        lib.callback.await('sm_barbershop:service:cancel', false)

        return notify('error', 'Pelanggan tidak terlihat.')
    end

    local origin = sceneOrigin(entry)
    local me = cache.ped

    lib.requestAnimDict(origin.dict)

    ---@type BarberService
    local current = {
        mode = 'barber',
        origin = origin,
        keeper = me,
        networked = true,
        customerId = result.customerId,
        customerPed = ped,
        result = result,
        stage = 'entering',
    }

    service = current
    lockPed(me)
    approach(me, origin, 'keeper_base')

    if service ~= current then
        busy = false

        return
    end

    local grip = gripFor(origin)

    if service ~= current then
        busy = false

        return
    end

    current.scissors = createScissors(true, GetEntityCoords(me))

    if grip then
        FreezeEntityPosition(current.scissors, false)
        attachTo(current.scissors, me, GetPedBoneIndex(me, grip.bone), grip.pos, grip.rot, grip.order)
        current.attached = true
    end

    local ready = keeperEnter(current)

    busy = false

    if service ~= current then return end

    if not ready then
        notify('error', 'Animasi barber gagal dimulai, silakan coba lagi.')
        lib.callback.await('sm_barbershop:service:cancel', false)

        return
    end

    current.stage = 'service'
    openServiceMenu(current, result, ped)
end

    local opened = appearance:openBarberMenu({

---@param entry BarberChairEntry
local function startSelfService(entry)
    if customer or service or busy or exiting then return end

    local state = Sync.shop(entry.shop)

    if not state then return end

    local answer = lib.alertDialog({
        header = state.label,
        content = ('Harga dihitung per item yang diubah, %s%% dari harga barber. Bayar tunai atau bank setelah selesai. Lanjutkan?'):format(state.selfServiceRate or 100),
        centered = true,
        cancel = true,
        labels = { cancel = 'Batal', confirm = 'Lanjut' },
    })

    if answer ~= 'confirm' then return end

    busy = true

    local result = lib.callback.await('sm_barbershop:selfservice:start', false, entry.shop, entry.index)

    if not result.success then
        busy = false

        return notify('error', result.message)
    end

    local origin = sceneOrigin(entry)

    lib.requestAnimDict(origin.dict)

    ---@type BarberSeat
    local seat = {
        shop = entry.shop,
        chair = entry.index,
        origin = origin,
        seated = pick(origin.set.seated),
        stage = 'entering',
    }

    local npc, scissors = spawnKeeper(entry.shop, origin)

    ---@type BarberService
    local current = { mode = 'self', origin = origin, keeper = npc, networked = false, scissors = scissors, result = result, stage = 'entering' }

    customer = seat
    service = current

    CreateThread(function()
        keeperWalk(current)
    end)

    lockPed(cache.ped)
    approach(cache.ped, origin, 'player_enterchair')
    CreateThread(seatLoop)

    while service == current and not current.ready do
        Wait(0)
    end

    if service ~= current then
        busy = false

        return
    end

    Citizen.CreateThreadNow(function()
        current.arrived = keeperEnter(current, 'keeper_enterchair')
    end)

    sitScene(seat)

    busy = false

    if service ~= current then return end

    seat.stage = 'self'

    while service == current and current.arrived == nil do
        Wait(100)
    end

    if service ~= current then return end

    current.stage = 'edit'

    openSelfMenu(current, seat, result)
end

---@param current BarberService
---@param deadline integer
local function cuttingLoop(current, deadline)
    local origin = current.origin

    current.entered = true

    if current.scissors and DoesEntityExist(current.scissors) then
        ResetEntityAlpha(current.scissors)
    end

    while service == current and GetGameTimer() < deadline do
        local clip = pick(origin.set.cutting)
        local finish = math.min(deadline, GetGameTimer() + math.max(MIN_CUT_MS, clipDuration(origin, clip)))

        playKeeper(current, clip, KEEPER_STAGE.cut)

        while service == current and GetGameTimer() < finish do
            Wait(100)
        end
    end

    if service == current then
        playKeeper(current, 'keeper_base', KEEPER_STAGE.settle)
    end
end

---@param duration number
local function cuttingProgress(duration)
    lib.progressCircle({
        duration = duration,
        label = 'Sedang mencukur',
        position = 'bottom',
        canCancel = false,
        disable = { move = true, combat = true, car = true },
    })
end

local function idle()
    return not (customer or service or busy or exiting)
end

---@param entry BarberChairEntry
local function pickupChair(entry)
    if not idle() then return end

    busy = true

    local result = lib.callback.await('sm_barbershop:portable:pickup', false, entry.index)

    busy = false

    if result.success then
        notify('success', 'Kursi cukur diangkat.')
    else
        notify('error', result.message)
    end
end

local config = Config
local limits = config.limits
local appearance = exports.sm_appearance

local PAID_RESUME_DELAY = 1000
local SCISSORS = config.items.scissors
local PREPARE_TIMEOUT = 8000
local SERVICE_REACH = 5.0

---@type table<string, table<number, BarberSession>>
local sessions = {}
---@type table<number, BarberSession>
local byCustomer = {}
---@type table<number, BarberSession>
local byBarber = {}
---@type table<number, table<string, number>>
local cooldowns = {}
local invoiceSequence = 0

---@type table<string, boolean>
local methods = {}

for i = 1, #config.payment.methods do
    methods[config.payment.methods[i]] = true
end

local function notify(source, notifyType, description)
    TriggerClientEvent('ox_lib:notify', source, {
        type = notifyType,
        description = description,
        duration = 4500,
    })
end

---@param message string
---@return BarberActionResult
local function fail(message)
    return { success = false, message = message }
end

local function playerExists(source)
    return DoesPlayerExist(tostring(source))
end

local function cooldown(source, key, ms)
    local now = GetGameTimer()
    local entry = cooldowns[source]

    if not entry then
        entry = {}
        cooldowns[source] = entry
    end

    if (entry[key] or 0) > now then
        return false
    end

    entry[key] = now + ms

    return true
end

---@param source number
---@param chair vector4
---@return boolean
local function nearChair(source, chair)
    local ped = GetPlayerPed(source)

    return ped ~= 0 and #(GetEntityCoords(ped) - chair.xyz) <= SERVICE_REACH
end

---@param shopId string
local function syncChairs(shopId)
    ---@type table<string, BarberChairStatus>
    local occupied = {}

    for index, session in pairs(sessions[shopId] or {}) do
        occupied[tostring(index)] = session.stage == 'waiting' and 'waiting' or 'busy'
    end

    Sync.setChairs(shopId, occupied)
end

---@param shop BarberShop
---@param chair number
---@param customer number
---@param stage BarberStage
---@return BarberSession
local function createSession(shop, chair, customer, stage)
    local list = sessions[shop.id]

    if not list then
        list = {}
        sessions[shop.id] = list
    end

    ---@type BarberSession
    local session = { shop = shop.id, chair = chair, customer = customer, stage = stage, mode = stage == 'self' and 'self' or 'barber' }

    list[chair] = session
    byCustomer[customer] = session
    syncChairs(shop.id)

    return session
end

---@param session BarberSession
local function detachBarber(session)
    if not session.barber then return end

    byBarber[session.barber] = nil
    session.barber = nil
end

---@param session BarberSession
---@param reason string
local function endSession(session, reason)
    if sessions[session.shop][session.chair] ~= session then return end

    sessions[session.shop][session.chair] = nil
    byCustomer[session.customer] = nil
    session.invoice = nil
    syncChairs(session.shop)

    if playerExists(session.customer) then
        TriggerClientEvent('sm_barbershop:session:ended', session.customer, reason)
    end

    if session.barber and playerExists(session.barber) then
        TriggerClientEvent('sm_barbershop:session:ended', session.barber, reason)
    end

    detachBarber(session)
    TriggerEvent('sm_barbershop:internal:sessionEnded', session.shop, session.chair)
end

---@param session BarberSession
---@param message string
local function releaseBarber(session, message)
    local barber = session.barber

    detachBarber(session)
    session.invoice = nil
    session.stage = 'waiting'
    session.original = nil
    session.prices = nil
    syncChairs(session.shop)

    if playerExists(session.customer) then
        TriggerClientEvent('sm_barbershop:customer:revert', session.customer)
        notify(session.customer, 'warning', message)
    end

    if barber and playerExists(barber) then
        TriggerClientEvent('sm_barbershop:session:ended', barber, 'released')
    end
end

---@param source number
---@return table? appearance
---@return string? reason
local function storedAppearance(source)
    if Player(source).state.forcedOutfit then
        return nil, 'Penampilan tidak bisa diubah saat memakai seragam.'
    end

    local stored = appearance:getAppearance(source)

    if type(stored) ~= 'table' or type(stored.hair) ~= 'table' or type(stored.headOverlays) ~= 'table' then
        return nil, 'Data penampilan tidak ditemukan.'
    end

    return stored
end

---@param player table
---@param source number
---@param method string
---@param amount integer
---@param final table
---@param reason string
---@return BarberActionResult
local function chargeAndSave(player, source, method, amount, final, reason)
    if amount > 0 and player.Functions.RemoveMoney(method, amount, reason) ~= true then
        return fail(method == 'cash' and 'Uang tunai tidak cukup.' or 'Saldo bank tidak cukup.')
    end

    if appearance:saveHead(source, Pricing.barberFields(final)) ~= true then
        if amount > 0 then
            player.Functions.AddMoney(method, amount, reason .. '-refund')
        end

        return fail('Penampilan gagal disimpan, uang dikembalikan.')
    end

    return { success = true }
end

---@param value any
---@return string?
local function methodOf(value)
    return type(value) == 'string' and methods[value] and value or nil
end

---@param target number
---@param baseline BarberAppearanceFields
---@return BarberCustomerSnapshot?
local function prepareCustomer(target, baseline)
    local result = promise.new()
    local settled = false

    local function settle(value)
        if settled then return end

        settled = true
        result:resolve(value)
    end

    lib.callback('sm_barbershop:customer:prepare', target, settle, baseline)
    SetTimeout(PREPARE_TIMEOUT, function() settle(false) end)

    return Citizen.Await(result) or nil
end

---@param session BarberSession
local function resumeSelf(session)
    session.stage = 'self'

    if playerExists(session.customer) then
        TriggerClientEvent('sm_barbershop:selfservice:resume', session.customer, Pricing.barberFields(session.original))
    end
end

---@param session BarberSession
local function resumeService(session)
    local customer, barber = session.customer, session.barber

    if not barber or not playerExists(barber) then return end

    session.stage = 'preparing'
    session.invoice = nil

    local snapshot = prepareCustomer(customer, Pricing.barberFields(session.original))

    if byCustomer[customer] ~= session or session.barber ~= barber then return end

    if type(snapshot) ~= 'table' or type(snapshot.appearance) ~= 'table' then
        return releaseBarber(session, 'Pelanggan tidak merespons, layanan dihentikan.')
    end

    session.stage = 'service'

    TriggerClientEvent('sm_barbershop:service:resume', barber, snapshot)
end

---@param session BarberSession
local function resume(session)
    if session.mode == 'self' then
        resumeSelf(session)
    else
        resumeService(session)
    end
end

---@param session BarberSession
---@param cut boolean
local function startCutting(session, cut)
    local customer = session.customer
    local duration = cut and limits.cutDuration or PAID_RESUME_DELAY

    session.stage = 'cutting'
    session.invoice = nil

    if cut then
        TriggerClientEvent('sm_barbershop:session:cutting', customer, duration)

        if session.barber then
            TriggerClientEvent('sm_barbershop:session:cutting', session.barber, duration)
        end
    end

    SetTimeout(duration, function()
        if byCustomer[customer] == session and session.stage == 'cutting' then
            resume(session)
        end
    end)
end

---@param session BarberSession
---@param message string
local function declineInvoice(session, message)
    session.invoice = nil

    if playerExists(session.customer) then
        notify(session.customer, 'warning', message)
    end

    CreateThread(function()
        resumeService(session)
    end)
end

---@param session BarberSession
---@param source number
---@return string?
local function serviceBlocked(session, source)
    if not Wear.slot(source, SCISSORS) then return 'Anda harus membawa gunting cukur yang masih bisa dipakai.' end
    if not Portable.usable(session.shop, session.chair) then return 'Kursi cukur ini sudah habis pakai.' end
end

lib.callback.register('sm_barbershop:chair:sit', function(source, shopId, chairIndex)
    if not cooldown(source, 'chair', 1000) then return fail('Tunggu sebentar.') end

    local shop = Shops.get(shopId)

    if not shop or not Shops.chair(shop, chairIndex) then return fail('Kursi tidak ditemukan.') end
    if byCustomer[source] then return fail('Anda sudah berada di kursi.') end
    if byBarber[source] then return fail('Anda sedang melayani pelanggan.') end
    if not Shops.isOperational(shop) then return fail('Tidak ada barber yang bertugas.') end
    if sessions[shop.id] and sessions[shop.id][chairIndex] then return fail('Kursi sedang dipakai.') end
    if not Portable.usable(shop.id, chairIndex) then return fail('Kursi cukur ini sudah habis pakai.') end

    createSession(shop, chairIndex, source, 'seating')

    return { success = true }
end)

lib.callback.register('sm_barbershop:chair:seated', function(source)
    local session = byCustomer[source]

    if not session or session.stage ~= 'seating' then return false end

    session.stage = 'waiting'
    syncChairs(session.shop)

    return true
end)

lib.callback.register('sm_barbershop:selfservice:start', function(source, shopId, chairIndex)
    if not cooldown(source, 'chair', 1000) then return fail('Tunggu sebentar.') end
    if type(chairIndex) ~= 'number' then return fail('Self service hanya tersedia di kursi toko.') end

    local shop = Shops.get(shopId)

    if not shop or not Shops.chair(shop, chairIndex) then return fail('Kursi tidak ditemukan.') end
    if byCustomer[source] then return fail('Anda sudah berada di kursi.') end
    if not Shops.selfServiceAllowed(shop) then return fail('Barber sedang bertugas, silakan duduk untuk dilayani.') end
    if sessions[shop.id] and sessions[shop.id][chairIndex] then return fail('Kursi sedang dipakai.') end

    local stored, reason = storedAppearance(source)

    if not stored then return fail(reason --[[@as string]]) end

    local session = createSession(shop, chairIndex, source, 'self')

    session.original = stored
    session.prices = Shops.priceList(shop, 'self')

    SetTimeout(limits.selfServiceTimeout * 1000, function()
        if byCustomer[source] == session and session.stage == 'self' then
            endSession(session, 'timeout')
        end
    end)

    return {
        success = true,
        chair = shop.chairs[chairIndex],
        prices = session.prices,
        appearance = Pricing.barberFields(stored),
        shopLabel = shop.label,
        customerName = Staff.nameOf(Staff.getPlayer(source)),
    }
end)

lib.callback.register('sm_barbershop:selfservice:pay', function(source, values, method)
    local session = byCustomer[source]

    if not session or session.stage ~= 'self' or not session.original then
        return fail('Sesi self service tidak ditemukan.')
    end

    method = methodOf(method)

    if not method then return fail('Metode pembayaran tidak valid.') end

    local final, items, total = Pricing.apply(session.original, Pricing.sanitize(values), session.prices)

    if #items == 0 then return fail('Belum ada perubahan yang dibayar.') end

    local shop = Shops.get(session.shop)
    local player = Staff.getPlayer(source)

    if not shop or not player then return fail('Data pemain tidak ditemukan.') end

    session.stage = 'paying'

    local result = chargeAndSave(player, source, method, total, final, 'barbershop-selfservice')

    if byCustomer[source] ~= session then return fail('Sesi self service sudah berakhir.') end

    if not result.success then
        session.stage = 'self'

        return result
    end

    local citizenId = Staff.citizenIdOf(player)
    local name = Staff.nameOf(player)

    if Shops.isOwned(shop) and total > 0 then
        Shops.addBalance(shop, total)
        Storage.addTransaction({
            shopId = shop.id,
            type = 'service',
            amount = total,
            balanceAfter = shop.balance,
            actorCitizenId = citizenId,
            actorName = name,
            note = ('Self service Â· %d item'):format(#items),
        })
    end

    Storage.addService({
        shopId = shop.id,
        mode = 'self',
        customerCitizenId = citizenId,
        customerName = name,
        amount = total,
        items = Pricing.describe(items),
    })

    TriggerEvent('sm_barbershop:internal:serviceRecorded', shop.id)
    session.original = final
    startCutting(session, Pricing.needsCut(items))

    return { success = true }
end)

lib.callback.register('sm_barbershop:chair:leave', function(source)
    local session = byCustomer[source]

    if not session then return true end

    if session.stage == 'cutting' or session.stage == 'paying' then
        return false
    end

    local barber = session.barber

    endSession(session, 'left')

    if barber and playerExists(barber) then
        notify(barber, 'warning', 'Pelanggan meninggalkan kursi.')
    end

    return true
end)

lib.callback.register('sm_barbershop:service:start', function(source, shopId, chairIndex)
    if not cooldown(source, 'service', 1000) then return fail('Tunggu sebentar.') end

    local shop = Shops.get(shopId)
    local chair = shop and Shops.chair(shop, chairIndex)

    if not shop or not chair then return fail('Kursi tidak ditemukan.') end

    local session = sessions[shop.id] and sessions[shop.id][chairIndex]

    if not session or session.stage ~= 'waiting' then
        return fail('Tidak ada pelanggan yang menunggu di kursi ini.')
    end

    if byBarber[source] or byCustomer[source] then
        return fail('Anda sedang dalam sesi lain.')
    end

    if not Staff.isOnDuty(source, shop.id) then
        return fail('Anda tidak bertugas di barbershop ini.')
    end

    if not nearChair(source, chair) then
        return fail('Anda terlalu jauh dari kursi.')
    end

    local blocked = serviceBlocked(session, source)

    if blocked then return fail(blocked) end

    local customerId = session.customer
    local customer = Staff.getPlayer(customerId)
    local barber = Staff.getPlayer(source)

    if not customer or not barber then
        return fail('Data pemain tidak ditemukan.')
    end

    local stored, reason = storedAppearance(customerId)

    if not stored then return fail(reason --[[@as string]]) end

    session.stage = 'preparing'
    session.barber = source
    byBarber[source] = session
    syncChairs(shop.id)

    local snapshot = prepareCustomer(customerId, Pricing.barberFields(stored))

    if byCustomer[customerId] ~= session or session.barber ~= source then
        return fail('Pelanggan sudah meninggalkan kursi.')
    end

    if type(snapshot) ~= 'table' or type(snapshot.appearance) ~= 'table' then
        detachBarber(session)
        session.stage = 'waiting'
        syncChairs(shop.id)

        return fail('Pelanggan tidak merespons.')
    end

    session.stage = 'service'
    session.original = stored
    session.prices = Shops.priceList(shop, 'barber')

    return {
        success = true,
        customerId = customerId,
        customerName = Staff.nameOf(customer),
        shopLabel = shop.label,
        prices = session.prices,
        snapshot = snapshot,
    }
end)

lib.callback.register('sm_barbershop:service:finish', function(source, values)
    local session = byBarber[source]

    if not session or session.stage ~= 'service' or not session.original then
        return fail('Sesi layanan tidak ditemukan.')
    end

    local shop = Shops.get(session.shop)
    local barber = Staff.getPlayer(source)

    if not shop or not barber then
        return fail('Data tidak ditemukan.')
    end

    local final, items, total = Pricing.apply(session.original, Pricing.sanitize(values), session.prices)

    if #items == 0 then return fail('Belum ada perubahan untuk ditagih.') end

    local blocked = serviceBlocked(session, source)

    if blocked then return fail(blocked) end

    invoiceSequence += 1

    local invoiceId = ('barber-%s-%s'):format(os.time(), invoiceSequence)
    local timeout = limits.invoiceTimeout

    session.stage = 'invoice'
    session.invoice = {
        id = invoiceId,
        amount = total,
        items = items,
        final = final,
        expiresAt = os.time() + timeout,
    }

    TriggerClientEvent('sm_barbershop:customer:invoice', session.customer, {
        id = invoiceId,
        amount = total,
        items = Pricing.describe(items),
        methods = config.payment.methods,
        shopLabel = shop.label,
        barberName = Staff.nameOf(barber),
        expiresAt = session.invoice.expiresAt,
        timeout = timeout,
    })

    SetTimeout(timeout * 1000, function()
        if byCustomer[session.customer] ~= session then return end
        if session.stage ~= 'invoice' or not session.invoice or session.invoice.id ~= invoiceId then return end

        if session.barber and playerExists(session.barber) then
            notify(session.barber, 'warning', 'Pelanggan tidak merespons tagihan.')
        end

        declineInvoice(session, 'Tagihan kedaluwarsa, penampilan dikembalikan.')
    end)

    return { success = true }
end)

lib.callback.register('sm_barbershop:invoice:pay', function(source, invoiceId, method)
    local session = byCustomer[source]
    local invoice = session and session.invoice

    if not session or session.stage ~= 'invoice' or not invoice or invoice.id ~= invoiceId then
        return fail('Tagihan tidak ditemukan.')
    end

    if os.time() >= invoice.expiresAt then
        declineInvoice(session, 'Tagihan kedaluwarsa, penampilan dikembalikan.')

        return fail('Tagihan sudah kedaluwarsa.')
    end

    method = methodOf(method)

    if not method then return fail('Metode pembayaran tidak valid.') end

    local shop = Shops.get(session.shop)
    local customer = Staff.getPlayer(source)
    local barber = session.barber and Staff.getPlayer(session.barber)

    if not shop or not customer or not barber then
        return fail('Barber tidak tersedia.')
    end

    session.stage = 'paying'

    local amount = invoice.amount
    local result = chargeAndSave(customer, source, method, amount, invoice.final, 'barbershop-service')

    if byCustomer[source] ~= session then return fail('Sesi sudah berakhir.') end

    if not result.success then
        session.stage = 'invoice'

        return result
    end

    local barberCitizenId = Staff.citizenIdOf(barber)
    local barberName = Staff.nameOf(barber)
    local customerName = Staff.nameOf(customer)
    local commission = 0

    if shop.ownerCitizenId ~= barberCitizenId then
        commission = math.floor(amount * shop.commissionRate / 100)

        if commission > 0 then
            barber.Functions.AddMoney('bank', commission, 'barbershop-commission')
        end
    end

    local net = amount - commission

    Shops.addBalance(shop, net)
    Storage.addTransaction({
        shopId = shop.id,
        type = 'service',
        amount = net,
        balanceAfter = shop.balance,
        actorCitizenId = barberCitizenId,
        actorName = barberName,
        note = ('Layanan untuk %s Â· %d item'):format(customerName, #invoice.items),
    })

    if commission > 0 then
        Storage.addTransaction({
            shopId = shop.id,
            type = 'commission',
            amount = commission,
            balanceAfter = shop.balance,
            actorCitizenId = barberCitizenId,
            actorName = barberName,
            note = ('Komisi layanan untuk %s'):format(customerName),
        })
    end

    Storage.addService({
        shopId = shop.id,
        mode = 'barber',
        customerCitizenId = Staff.citizenIdOf(customer),
        customerName = customerName,
        employeeCitizenId = barberCitizenId,
        employeeName = barberName,
        amount = amount,
        commission = commission,
        items = Pricing.describe(invoice.items),
    })

    TriggerEvent('sm_barbershop:internal:serviceRecorded', shop.id)

    if commission > 0 then
        notify(session.barber, 'success', ('Komisi $%s masuk ke rekening bank.'):format(commission))
    end

    session.original = invoice.final

    local scissorsLeft = Wear.consume(session.barber, SCISSORS)
    local chairLeft = Portable.consume(session.shop, session.chair)

    if scissorsLeft then
        notify(session.barber, scissorsLeft > 0 and 'inform' or 'warning',
            scissorsLeft > 0 and ('Sisa pemakaian gunting cukur: %d.'):format(scissorsLeft) or 'Gunting cukur Anda sudah habis pakai.')
    end

    if chairLeft then
        notify(session.barber, chairLeft > 0 and 'inform' or 'warning',
            chairLeft > 0 and ('Sisa pemakaian kursi cukur: %d.'):format(chairLeft) or 'Kursi cukur sudah habis pakai, akan dibereskan setelah pelanggan berdiri.')
    end

    startCutting(session, Pricing.needsCut(invoice.items))

    return { success = true }
end)

local config = Config
local limits = config.limits

local Shops = { list = {}, byId = {} }

---@param key string
---@param value any
---@return integer?
local function itemPrice(key, value)
    local price = tonumber(value)
    local min, max = Pricing.bounds(key)

    if not price or price ~= price then return end

    price = math.floor(price)

    if price < min or price > max then return end

    return price
end

---@param key string
---@param value any
---@return integer
local function storedPrice(key, value)
    local price = tonumber(value)

    if not price or price ~= price then return Pricing.defaultPrice(key) end

    local min, max = Pricing.bounds(key)

    return math.min(max, math.max(min, math.floor(price)))
end

---@param value any
---@return integer?
local function selfServiceRate(value)
    local rate = tonumber(value)

    if not rate or rate ~= rate then return end

    rate = math.floor(rate)

    if rate < limits.minSelfServiceRate or rate > limits.maxSelfServiceRate then return end

    return rate
end

---@param value any
---@return string?
local function shopName(value)
    return Text.name(value, limits.minShopName, limits.maxShopName)
end

---@param raw any
---@return table<string, integer>
local function decodePrices(raw)
    local ok, decoded = pcall(json.decode, type(raw) == 'string' and raw or 'null')
    local stored = ok and type(decoded) == 'table' and decoded or {}
    local prices = {}

    for i = 1, #config.priceItems do
        local item = config.priceItems[i]

        prices[item.key] = storedPrice(item.key, stored[item.key])
    end

    return prices
end

function Shops.init()
    local rows = Storage.loadShops()

    for i = 1, #config.shops do
        local entry = config.shops[i]
        local row = rows[entry.id]

        ---@type BarberShop
        local shop = {
            id = entry.id,
            label = shopName(row and row.name) or entry.label,
            defaultLabel = entry.label,
            chairs = entry.coords.chairs,
            purchasePrice = config.purchesable_shop.price,
            purchesable = config.purchesable_shop.enable == true,
            ownerCitizenId = row and row.owner_citizenid or nil,
            ownerName = row and row.owner_name or nil,
            balance = row and math.floor(tonumber(row.balance) or 0) or 0,
            commissionRate = row and math.floor(tonumber(row.commission_rate) or 0) or config.defaults.commissionRate,
            selfServiceRate = selfServiceRate(row and row.self_service_rate) or config.defaults.selfServiceRate,
            itemPrices = decodePrices(row and row.item_prices),
        }

        if not row then
            Storage.saveShop(shop)
        end

        Shops.list[#Shops.list + 1] = shop
        Shops.byId[shop.id] = shop
    end
end

---@param shop BarberShop
---@param delta number
---@return number balance
function Shops.addBalance(shop, delta)
    shop.balance = math.max(0, shop.balance + math.floor(delta))
    Shops.save(shop)

    return shop.balance

---@param shop BarberShop
---@param payload { itemPrices: any, commissionRate: any, selfServiceRate: any }
---@return boolean
function Shops.setPrices(shop, payload)
    local commissionRate = tonumber(payload.commissionRate)
    local rate = selfServiceRate(payload.selfServiceRate)

    if not commissionRate or commissionRate ~= commissionRate or not rate or type(payload.itemPrices) ~= 'table' then
        return false
    end

    commissionRate = math.floor(commissionRate)

    if commissionRate < 0 or commissionRate > limits.maxCommissionRate then return false end

    local prices = {}

    for i = 1, #config.priceItems do
        local key = config.priceItems[i].key
        local price = itemPrice(key, payload.itemPrices[key])

        if not price then return false end

        prices[key] = price
    end

    shop.itemPrices = prices
    shop.commissionRate = commissionRate
    shop.selfServiceRate = rate
    Shops.save(shop)

    return true
end

---@param shop BarberShop
---@param mode 'barber'|'self'
---@return table<string, integer>
function Shops.priceList(shop, mode)
    local list = {}

    for key, price in pairs(shop.itemPrices) do
        list[key] = mode == 'self' and math.floor(price * shop.selfServiceRate / 100 + 0.5) or price
    end

    return list
end

---@return BarberShopPublic
function Shops.publicData(shop)
    return {
        id = shop.id,
        label = shop.label,
        defaultLabel = shop.defaultLabel,
        owned = Shops.isOwned(shop),
        ownerName = shop.ownerName,
        balance = shop.balance,
        commissionRate = shop.commissionRate,
        selfServiceRate = shop.selfServiceRate,
        itemPrices = shop.itemPrices,
        purchasePrice = shop.purchasePrice,
    }
end

    local charinfo = player and player.PlayerData and player.PlayerData.charinfo

    if charinfo and (charinfo.firstname or charinfo.lastname) then
        local full = ('%s %s'):format(charinfo.firstname or '', charinfo.lastname or '')

        return (full:gsub('^%s+', ''):gsub('%s+$', ''))
    end

    return fallback or 'Tidak diketahui'

    if not citizenId then return end

    local shop = Shops.get(shopId)

function Staff.permissions(shopId, role)
    local owner = role == 'owner'
    local granted = not owner and Roles.permissions(shopId, role) or {}
    local set = { roles = owner }

    for i = 1, #config.permissions do
        local key = config.permissions[i].key

        set[key] = owner or granted[key] == true
    end

    return set
end

---@param shopId string
---@param actorRole string?
---@param targetRole string?
---@return boolean
function Staff.outranks(shopId, actorRole, targetRole)
    if not actorRole or not targetRole or targetRole == 'owner' then return false end
    if actorRole == 'owner' then return true end

    local actor = Staff.permissions(shopId, actorRole)
    local target = Staff.permissions(shopId, targetRole)
    local wider = false

    for key, granted in pairs(actor) do
        if target[key] and not granted then return false end

        if granted and not target[key] then
            wider = true
        end
    end

    return wider
end

            name = employee.name,
            role = employee.role,
            hiredAt = employee.hiredAt,
            onDuty = onDuty[citizenId] == true,
            services = stat and stat.services or 0,
            revenue = stat and stat.revenue or 0,
        }
    end

    local weight = {}

    for _, role in ipairs(Roles.list(shopId)) do
        weight[role.key] = #Roles.permissionList(role)
    end

    table.sort(list, function(a, b)
        local left, right = weight[a.role] or 0, weight[b.role] or 0

        if left ~= right then
            return left > right
        end

        return a.name < b.name
    end)

    return list
end

---@param source number
---@param shopId string
---@return boolean
function Staff.isOnDuty(source, shopId)
    return Staff.duty[source] == shopId
end

---@param source number
function Staff.clearDuty(source)
    local shopId = Staff.duty[source]

    if not shopId then return end

    Staff.duty[source] = nil

    local shop = Staff.dutyByShop[shopId]

    if shop then
        shop[source] = nil
    end

    Player(source).state:set('barberDuty', false, true)

    local record = Shops.get(shopId)

    if record then
        Shops.sync(record)
    end
end

---@param source number
---@param shopId string
---@return boolean onDuty
function Staff.toggleDuty(source, shopId)
    if Staff.isOnDuty(source, shopId) then
        Staff.clearDuty(source)

        return false
    end

    local citizenId = Staff.citizenId(source)

    if not citizenId then
        return false
    end

    Staff.clearDuty(source)
    Staff.duty[source] = shopId

    local shop = Staff.dutyByShop[shopId]

    if not shop then
        shop = {}
        Staff.dutyByShop[shopId] = shop
    end

    shop[source] = citizenId
    Player(source).state:set('barberDuty', shopId, true)

    local record = Shops.get(shopId)

    if record then
        Shops.sync(record)
    end

    return true
end

---@param shopId string
function Staff.enforceDuty(shopId)
    for source in pairs(Staff.dutyByShop[shopId] or {}) do
        if not Staff.can(source, shopId, 'duty') then
            Staff.clearDuty(source)
            TriggerClientEvent('ox_lib:notify', source, {
                type = 'warning',
                description = 'Shift Anda diakhiri karena role Anda tidak lagi punya izin shift.',
                duration = 4500,
            })
        end
    end
end

---@param shopId string
---@return number
function Staff.dutyCount(shopId)
    local total = 0

    for _ in pairs(Staff.dutyByShop[shopId] or {}) do
        total += 1
    end

    return total
end

end)

local Sync = {}

---@type table<string, BarberShopSync>
local entries = {}

---@type BarberChairEntry[]
local portables = {}

---@type fun(list: BarberChairEntry[])[]
local listeners = {}

---@type fun(shopId: string, state: BarberShopState)[]
local shopListeners = {}

---@param shopId string
---@return BarberShopState?
function Sync.shop(shopId)
    local entry = entries[shopId]

    return entry and entry.state
end

---@param shopId string
---@param index number|string
---@return BarberChairStatus?
function Sync.chair(shopId, index)
    local entry = entries[shopId]

    return entry and entry.chairs[tostring(index)] or nil
end

---@return BarberChairEntry[]
function Sync.portables()
    return portables
end

---@param listener fun(shopId: string, state: BarberShopState)
function Sync.onShop(listener)
    shopListeners[#shopListeners + 1] = listener

    for shopId, entry in pairs(entries) do
        if entry.state then listener(shopId, entry.state) end
    end
end

---@param listener fun(list: BarberChairEntry[])
function Sync.onPortables(listener)
    listeners[#listeners + 1] = listener
    listener(portables)
end

local function rebuildPortables()
    ---@type BarberChairEntry[]
    local list = {}

    for shopId, entry in pairs(entries) do
        local placed = entry.portables or {}

        for i = 1, #placed do
            local chair = placed[i]

            list[#list + 1] = { shop = shopId, index = chair.id, coords = chair.coords, portable = true, ownerId = chair.ownerId }
        end
    end

    portables = list

    for i = 1, #listeners do
        listeners[i](list)
    end
end

RegisterNetEvent('sm_barbershop:sync', function(list)
    if source == '' or type(list) ~= 'table' then return end

    for i = 1, #list do
        local entry = list[i]
        local current = entries[entry.id]

        if not current or entry.revision > current.revision then
            entries[entry.id] = entry

            if entry.state then
                for j = 1, #shopListeners do
                    shopListeners[j](entry.id, entry.state)
                end
            end
        end
    end

    rebuildPortables()
end)

TriggerServerEvent('sm_barbershop:sync:request')

return Sync

local EVENT = 'sm_barbershop:sync'

local Sync = { revision = 0 }

---@type table<string, BarberShopSync>
local entries = {}

---@type table<number, boolean>
local served = {}

local started = false

---@return integer
function Sync.nextRevision()
    Sync.revision += 1

    return Sync.revision
end

---@param a table?
---@param b table?
---@return boolean
local function same(a, b)
    if type(a) ~= 'table' or type(b) ~= 'table' then return a == b end

    for key, value in pairs(a) do
        if b[key] ~= value then return false end
    end

    for key in pairs(b) do
        if a[key] == nil then return false end
    end

    return true
end

---@param shopId string
---@return BarberShopSync
local function entryOf(shopId)
    local entry = entries[shopId]

    if not entry then
        entry = { id = shopId, revision = 0, chairs = {}, portables = {} }
        entries[shopId] = entry
    end

    return entry
end

---@param entry BarberShopSync
local function publish(entry)
    entry.revision = Sync.nextRevision()

    if started then
        lib.latent.client(EVENT, -1, { entry })
    end
end

---@return BarberShopSync[]
local function snapshot()
    local list = {}

    for _, entry in pairs(entries) do
        list[#list + 1] = entry
    end

    return list
end

---@param shopId string
---@param state BarberShopState
function Sync.setState(shopId, state)
    local entry = entryOf(shopId)

    if same(entry.state, state) then return end

    entry.state = state
    publish(entry)
end

---@param shopId string
---@param chairs table<string, BarberChairStatus>
function Sync.setChairs(shopId, chairs)
    local entry = entryOf(shopId)

    if same(entry.chairs, chairs) then return end

    entry.chairs = chairs
    publish(entry)
end

---@param shopId string
---@param portables BarberPortableSync[]
function Sync.setPortables(shopId, portables)
    local entry = entryOf(shopId)

    entry.portables = portables
    publish(entry)
end

---@param shopId string
---@param index number|string
---@return BarberChairStatus?
function Sync.chairStatus(shopId, index)
    local entry = entries[shopId]

    return entry and entry.chairs[tostring(index)] or nil
end

function Sync.start()
    started = true
    lib.latent.client(EVENT, -1, snapshot())
end

RegisterNetEvent('sm_barbershop:sync:request', function()
    local source = source

    if not started or served[source] then return end

    served[source] = true
    lib.latent.client(EVENT, source, snapshot())
end)

AddEventHandler('playerDropped', function()
    served[source] = nil
end)

return Sync

local config = Config
local inventory = exports.ox_inventory

local DEFAULT_USES = 3

local Wear = {}

---@type table<string, integer>
local maxUses = {}

for i = 1, #config.supplies do
    maxUses[config.supplies[i].item] = config.supplies[i].uses or DEFAULT_USES
end

---@param item string
---@return integer
function Wear.max(item)
    return maxUses[item] or DEFAULT_USES
end

---@param metadata table?
---@return integer
function Wear.remaining(metadata)
    local used = metadata and tonumber(metadata.used)

    return used and math.floor(used) or 0
end

---@param item string
---@param used integer
---@param base table?
---@return table
function Wear.metadata(item, used, base)
    local metadata = base and lib.table.deepclone(base) or {}

    metadata.used = used
    metadata.description = ('Sisa pemakaian: %d/%d'):format(used, Wear.max(item))

    return metadata
end

---@param source number
---@param item string
---@return table? slot
function Wear.slot(source, item)
    local slots = inventory:Search(source, 'slots', item)
    local best, bestLeft

    if type(slots) ~= 'table' then return end

    for i = 1, #slots do
        local left = Wear.remaining(slots[i].metadata)

        if left > 0 and (not bestLeft or left < bestLeft) then
            best, bestLeft = slots[i], left
        end
    end

    return best
end

---@param source number
---@param item string
---@return integer? left
function Wear.consume(source, item)
    local slot = Wear.slot(source, item)

    if not slot then return end

    local left = Wear.remaining(slot.metadata) - 1

    if left <= 0 then
        inventory:RemoveItem(source, item, 1, nil, slot.slot)

        return 0
    end

    inventory:SetMetadata(source, slot.slot, Wear.metadata(item, left, slot.metadata))

    return left
end

return Wear
